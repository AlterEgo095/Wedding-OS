def migrate_table(sqlite_cur, pg_cur, table):
    sqlite_cur.execute(f"SELECT * FROM {table}")
    rows = sqlite_cur.fetchall()
    cols = [d[0] for d in sqlite_cur.description]
    # Bulk insert with execute_values for speed
    insert_sql = f'INSERT INTO "{table}" ({",".join(cols)}) VALUES %s'
    execute_values(pg_cur, insert_sql, rows, template=None, page_size=1000)
    return len(rows)

#!/usr/bin/env bash
# pg-backup.sh — V5.0 PostgreSQL daily backup

OUT="$BACKUP_DIR/wedding-pg-$TS.sql.gz"
