   215→// ════════════════════════════════════════════════════════════════════════════
   216→// usePlatformFetch — wraps fetch with Bearer token + session-expiry handling
   217→// ════════════════════════════════════════════════════════════════════════════
   218→
   219→function usePlatformFetch() {
   220→  const router = useRouter()
   221→  const sessionExpiredRef = useRef(false)
   222→
   223→  const onSessionExpired = useCallback(() => {
   224→    if (sessionExpiredRef.current) return
   225→    sessionExpiredRef.current = true
   226→    try {
   227→      localStorage.removeItem('admin_token')
   228→      localStorage.removeItem('admin_user')
   229→    } catch {
   230→      /* ignore */
   231→    }
   232→    toast.error('Session expirée, veuillez vous reconnecter')
   233→    router.replace('/platform/login')
   234→  }, [router])
   235→
   236→  const getToken = useCallback(() => {
   237→    if (typeof window === 'undefined') return null
   238→    return localStorage.getItem('admin_token')
   239→  }, [])
   240→
   241→  const fetchWithAuth = useCallback(
   242→    async (url: string, init?: RequestInit): Promise<Response | null> => {
   243→      const token = getToken()
   244→      if (!token) {
   245→        onSessionExpired()
   246→        return null
   247→      }
   248→      let res: Response
   249→      try {
   250→        res = await fetch(url, {
   251→          ...init,
   252→          headers: {
   253→            ...(init?.headers || {}),
   254→            Authorization: `Bearer ${token}`,
   255→          },
   256→        })
   257→      } catch {
   258→        toast.error('Erreur de connexion au serveur')
   259→        return null
   260→      }
   261→      if (res.status === 401) {
   262→        onSessionExpired()
   263→        return null
   264→      }
   265→      if (res.status === 403) {
   266→        toast.error('Accès refusé')
   267→        return null
   268→      }
   269→      return res
   270→    },
   271→    [getToken, onSessionExpired]
   272→  )
   273→
   274→  return { fetchWithAuth, onSessionExpired, getToken }
   275→}

   305→function formatDate(iso: string | null): string {
   306→  if (!iso) return '—'
   307→  try {
   308→    return new Date(iso).toLocaleDateString('fr-FR', {
   309→      day: '2-digit',
   310→      month: 'short',
   311→      year: 'numeric',
   312→    })
   313→  } catch {
   314→    return '—'
   315→  }
   316→}
   317→
   318→function formatDateTime(iso: string | null): string {
   319→  if (!iso) return '—'
   320→  try {
   321→    return new Date(iso).toLocaleString('fr-FR', {
   322→      day: '2-digit',
   323→      month: '2-digit',
   324→      year: 'numeric',
   325→      hour: '2-digit',
   326→      minute: '2-digit',
   327→    })
   328→  } catch {
   329→    return '—'
   330→  }
   331→}
   332→
   333→function toDateInput(iso: string | null): string {
   334→  if (!iso) return ''
   335→  try {
   336→    return new Date(iso).toISOString().slice(0, 10)
   337→  } catch {
   338→    return ''
   339→  }
   340→}
