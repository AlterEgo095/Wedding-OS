   269→// usePlatformFetch — wraps fetch with Bearer token + session-expiry handling
   270→// ════════════════════════════════════════════════════════════════════════════
   271→
   272→function usePlatformFetch() {
   273→  const router = useRouter()
   274→  const sessionExpiredRef = useRef(false)
   275→
   276→  const onSessionExpired = useCallback(() => {
   277→    if (sessionExpiredRef.current) return
   278→    sessionExpiredRef.current = true
   279→    try {
   280→      localStorage.removeItem('admin_token')
   281→      localStorage.removeItem('admin_user')
   282→    } catch {
   283→      /* ignore */
   284→    }
   285→    toast.error('Session expirée, veuillez vous reconnecter')
   286→    router.replace('/platform/login')
   287→  }, [router])
   288→
   289→  const getToken = useCallback(() => {
   290→    if (typeof window === 'undefined') return null
   291→    return localStorage.getItem('admin_token')
   292→  }, [])
   293→
   294→  const fetchWithAuth = useCallback(
   295→    async (url: string, init?: RequestInit): Promise<Response | null> => {
   296→      const token = getToken()
   297→      if (!token) {
   298→        onSessionExpired()
   299→        return null
   300→      }
   301→      let res: Response
   302→      try {
   303→        res = await fetch(url, {
   304→          ...init,
   305→          headers: {
   306→            ...(init?.headers || {}),
   307→            Authorization: `Bearer ${token}`,
   308→          },
   309→        })
   310→      } catch {
   311→        toast.error('Erreur de connexion au serveur')
   312→        return null
   313→      }
   314→      if (res.status === 401) {
   315→        onSessionExpired()
   316→        return null
   317→      }
   318→      if (res.status === 403) {
   319→        toast.error('Accès refusé')
   320→        return null
   321→      }
   322→      return res
   323→    },
   324→    [getToken, onSessionExpired]
   325→  )
   326→
   327→  return { fetchWithAuth, onSessionExpired, getToken }
   328→}

   358→function formatDate(iso: string | null): string {
   359→  if (!iso) return '—'
   360→  try {
   361→    return new Date(iso).toLocaleDateString('fr-FR', {
   362→      day: '2-digit',
   363→      month: 'short',
   364→      year: 'numeric',
   365→    })
   366→  } catch {
   367→    return '—'
   368→  }
   369→}
   370→
   371→function formatDateTime(iso: string | null): string {
   372→  if (!iso) return '—'
   373→  try {
   374→    return new Date(iso).toLocaleString('fr-FR', {
   375→      day: '2-digit',
   376→      month: '2-digit',
   377→      year: 'numeric',
   378→      hour: '2-digit',
   379→      minute: '2-digit',
   380→    })
   381→  } catch {
   382→    return '—'
   383→  }
   384→}

   399→function DashboardTab({ fetchWithAuth }: { fetchWithAuth: (url: string, init?: RequestInit) => Promise<Response | null> }) {
   400→  const [data, setData] = useState<DashboardData | null>(null)
   401→  const [loading, setLoading] = useState(true)
   402→
   403→  const load = useCallback(async () => {
   404→    setLoading(true)
   405→    const res = await fetchWithAuth('/api/platform/dashboard')
   406→    if (!res) {
   407→      setLoading(false)
   408→      return
   409→    }
   410→    try {
   411→      const json = (await res.json()) as DashboardData
   412→      setData(json)
   413→    } catch {
   414→      toast.error('Réponse invalide du serveur')
   415→    } finally {
   416→      setLoading(false)
   417→    }
   418→  }, [fetchWithAuth])
   419→
   420→  useEffect(() => {
   421→    load()
   422→  }, [load])
   423→
   424→  if (loading) {
   425→    return (
   426→      <div className="space-y-6 p-4 md:p-6">
   427→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   428→          {Array.from({ length: 4 }).map((_, i) => (
   429→            <Skeleton key={i} className="h-28 rounded-xl" />
   430→          ))}
   431→        </div>
   432→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   433→          <Skeleton className="h-96 rounded-xl" />
   434→          <Skeleton className="h-96 rounded-xl" />
   435→        </div>
   436→      </div>
   437→    )
   438→  }
   439→
   440→  if (!data) return null
   441→

   472→    },
   473→    {

   767→function WeddingsTab({ fetchWithAuth }: { fetchWithAuth: (url: string, init?: RequestInit) => Promise<Response | null> }) {
   768→  const [weddings, setWeddings] = useState<Wedding[]>([])
   769→  const [total, setTotal] = useState(0)
   770→  const [totalPages, setTotalPages] = useState(1)
   771→  const [page, setPage] = useState(1)
   772→  const [loading, setLoading] = useState(true)
   773→  const [search, setSearch] = useState('')
   774→  const [statusFilter, setStatusFilter] = useState<string>('ALL')
   775→  const [planFilter, setPlanFilter] = useState<string>('ALL')
   776→
   777→  // Dialog state
   778→  const [showFormDialog, setShowFormDialog] = useState(false)
   779→  const [showDeleteDialog, setShowDeleteDialog] = useState(false)
   780→  const [editing, setEditing] = useState<Wedding | null>(null)
   781→  const [deleting, setDeleting] = useState<Wedding | null>(null)
   782→  const [form, setForm] = useState<WeddingFormState>(EMPTY_WEDDING_FORM)
   783→  const [saving, setSaving] = useState(false)
   784→
   785→  const LIMIT = 20
   786→  const searchRef = useRef(search)
   787→  searchRef.current = search
   788→
   789→  const load = useCallback(
   790→    async (targetPage: number) => {
   791→      setLoading(true)
   792→      const params = new URLSearchParams({
   793→        page: String(targetPage),
   794→        limit: String(LIMIT),
   795→      })
   796→      if (searchRef.current) params.set('search', searchRef.current)
   797→      if (statusFilter !== 'ALL') params.set('status', statusFilter)
   798→      if (planFilter !== 'ALL') params.set('plan', planFilter)
   799→
   800→      const res = await fetchWithAuth(`/api/platform/weddings?${params.toString()}`)
   801→      if (!res) {
   802→        setLoading(false)
   803→        return
   804→      }
   805→      try {
   806→        const json = (await res.json()) as PaginatedWeddings
   807→        setWeddings(json.weddings || [])
   808→        setTotal(json.total || 0)
   809→        // API returns { total, page, limit } — compute totalPages client-side.
   810→        setTotalPages(Math.max(1, Math.ceil((json.total || 0) / (json.limit || LIMIT))))
   811→        setPage(json.page || targetPage)
   812→      } catch {
   813→        toast.error('Réponse invalide du serveur')
   814→      } finally {
   815→        setLoading(false)
   816→      }
   817→    },
   818→    [fetchWithAuth, statusFilter, planFilter]
   819→  )
   820→
   821→  useEffect(() => {
   822→    load(1)
   823→  }, [statusFilter, planFilter, load])
   824→
   825→  // Debounced search trigger
   826→  useEffect(() => {
   827→    const t = setTimeout(() => {
   828→      load(1)
   829→    }, 350)
   830→    return () => clearTimeout(t)
   831→  }, [search, load])
   832→

   834→    setEditing(null)
   835→    setForm(EMPTY_WEDDING_FORM)
   836→    setShowFormDialog(true)
   837→  }
   838→
   839→  const openEdit = (w: Wedding) => {
   840→    setEditing(w)
   841→    setForm({
   842→      slug: w.slug,
   843→      brideName: w.brideName,
   844→      groomName: w.groomName,
   845→      weddingDate: toDateInput(w.weddingDate),
   846→      venueCity: w.venueCity || '',
   847→      status: w.status,
   848→      plan: w.plan,
   849→    })
   850→    setShowFormDialog(true)
   851→  }
   852→
   853→  const handleSave = async () => {
   854→    if (!form.slug || !form.brideName || !form.groomName) {
   855→      toast.error('Slug, mariée et marié sont requis')
   856→      return
   857→    }
   858→    setSaving(true)
   859→    const payload: Record<string, unknown> = {
   860→      slug: form.slug.trim().toLowerCase(),
   861→      brideName: form.brideName.trim(),
   862→      groomName: form.groomName.trim(),
   863→      weddingDate: form.weddingDate ? new Date(form.weddingDate).toISOString() : null,
   864→      venueCity: form.venueCity.trim() || null,
   865→      status: form.status,
   866→      plan: form.plan,
   867→    }
   868→    try {
   869→      const url = editing
   870→        ? `/api/platform/weddings/${editing.id}`
   871→        : '/api/platform/weddings'
   872→      const method = editing ? 'PUT' : 'POST'
   873→      const res = await fetchWithAuth(url, {
   874→        method,
   875→        headers: { 'Content-Type': 'application/json' },
   876→        body: JSON.stringify(payload),
   877→      })
   878→      if (!res) {
   879→        setSaving(false)
   880→        return
   881→      }
   882→      const json = await res.json().catch(() => ({}))
   883→      if (res.ok) {
   884→        toast.success(editing ? 'Mariage mis à jour' : 'Mariage créé')
   885→        setShowFormDialog(false)
   886→        setEditing(null)
   887→        setForm(EMPTY_WEDDING_FORM)
   888→        load(page)

   891→      }
   892→    } catch {
   893→      toast.error('Erreur de connexion')
   894→    } finally {
   895→      setSaving(false)
   896→    }
   897→  }
   898→
   899→  const handleDelete = async () => {

   913→        setShowDeleteDialog(false)
   914→        setDeleting(null)
   915→        load(1)
   916→      } else {
   917→        toast.error(json.error || 'Erreur lors de la suppression')
   918→      }
   919→    } catch {
   920→      toast.error('Erreur de connexion')
   921→    } finally {
   922→      setSaving(false)
   923→    }
   924→  }
   925→
   926→  return (
   927→    <div className="space-y-4 p-4 md:p-6">

   978→      <Card className="glass-card gold-border border-0">

   980→          <div className="max-h-[60vh] overflow-y-auto custom-scrollbar">

  1078→            </Table>
