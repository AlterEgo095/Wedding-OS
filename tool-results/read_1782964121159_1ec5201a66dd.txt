   179→function ModulePreview({ pack, ds, couple }: { pack: CollectionPack; ds: PremiumCollection['designSystem']; couple: CouplePreview }) {

   184→    <div className="space-y-3">

   186→        const chosenVariantId = activeVariantByModule[m.id] || m.variants[0]?.id
   187→        const chosenVariant = m.variants.find((v) => v.id === chosenVariantId) || m.variants[0]
   188→        const isExpanded = expandedModule === m.id
   189→        return (
   190→          <div key={m.id} className="rounded-xl overflow-hidden" style={{ background: ds.background, border: `1px solid ${ds.primary}25` }}>
   191→            <div className="p-3 flex items-center gap-3" style={{ background: ds.surface }}>

   197→                  <p className="text-sm font-semibold truncate" style={{ color: ds.text }}>{m.name}</p>
   198→                  {m.required && (

   271→function CollectionDetail({ collection, onClose }: { collection: PublicCollection; onClose: () => void }) {
   272→  const [activePack, setActivePack] = useState<PackId>('website')
   273→  const [fullCollection, setFullCollection] = useState<PremiumCollection | null>(null)
   274→  const [couple, setCouple] = useState<CouplePreview>(NEUTRAL_COUPLE)
   275→  // Phase D — deploy state
   276→  const [deploying, setDeploying] = useState(false)
   277→  const [deployResult, setDeployResult] = useState<{ ok: boolean; message: string } | null>(null)
   278→
   279→  // Phase B — fetch the current wedding's couple identity (tenant-aware).
   280→  // /api/settings resolves the wedding from X-Wedding-Slug header / auth / default
   281→  // and returns bride_name, groom_name, wedding_date, venue_name, hashtag.
   282→  useEffect(() => {
   283→    fetch('/api/settings')
   284→      .then((r) => (r.ok ? r.json() : null))
   285→      .then((data) => {
   286→        if (!data?.settings) return
   287→        const s = data.settings
   288→        const bride = (s.bride_name || '').trim()
   289→        const groom = (s.groom_name || '').trim()
   290→        const label = [bride, groom].filter(Boolean).join(' & ') || 'Mari & Mme'
   291→        const dateRaw = s.wedding_date || ''
   292→        const dateDisplay = dateRaw
   293→          ? new Date(dateRaw).toLocaleDateString('fr-FR', { day: 'numeric', month: 'long', year: 'numeric' })
   294→          : 'Date à définir'
   295→        setCouple({
   296→          bride: bride || 'Mme',
   297→          groom: groom || 'M.',
   298→          label,
   299→          date: dateDisplay,
   300→          venue: (s.venue_name || '').trim() || 'Lieu à définir',
   301→          hashtag: (s.hashtag || '').trim() || '',
   302→        })
   303→      })
   304→      .catch(() => {})
   305→  }, [])
   306→
   307→  useEffect(() => {
   308→    fetch(`/api/collections/${collection.id}`)
   309→      .then((r) => r.json())
   310→      .then((d) => setFullCollection(d.collection))
   311→      .catch(() => {})
   312→  }, [collection.id])
   313→
   314→  const ds = collection.designSystem
   315→  const pack = fullCollection?.packs.find((p) => p.id === activePack)
   316→
   317→  // Phase D — wire the "Déployer cette Collection" button to the deploy endpoint.
   318→  // The endpoint requires ORGANIZER+ auth and persists a WeddingCollectionBinding
   319→  // + applies the Collection's DesignSystem to the Wedding's Theme row.
   320→  const handleDeploy = async () => {
   321→    setDeploying(true)
   322→    setDeployResult(null)
   323→    try {
   324→      const res = await fetch('/api/collections/deploy', {
   325→        method: 'POST',
   326→        headers: { 'Content-Type': 'application/json' },
   327→        body: JSON.stringify({ collectionId: collection.id }),
   328→      })
   329→      const data = await res.json().catch(() => ({}))
   330→      if (res.ok && data?.success) {
   331→        setDeployResult({
   332→          ok: true,
   333→          message: `Collection "${collection.name}" déployée — le thème du mariage a été mis à jour.`,
   334→        })
   335→      } else if (res.status === 401) {
   336→        setDeployResult({ ok: false, message: 'Connexion requise — connectez-vous en tant qu\u2019Organisateur.' })
   337→      } else if (res.status === 403) {
   338→        setDeployResult({ ok: false, message: 'Permissions insuffisantes — rôle Organisateur requis.' })
   339→      } else {
   340→        setDeployResult({ ok: false, message: data?.error || 'Échec du déploiement.' })
   341→      }
   342→    } catch {
   343→      setDeployResult({ ok: false, message: 'Erreur réseau — réessayez.' })
   344→    } finally {
   345→      setDeploying(false)
   346→    }
   347→  }
   348→
   349→  return (
   350→    <DialogContent className="max-w-5xl max-h-[92vh] overflow-hidden flex flex-col p-0 gap-0" style={{ background: ds.background, border: `1px solid ${ds.primary}40` }}>
   351→      {/* Header */}
   352→      <div className="relative p-6 pb-4" style={{ background: `linear-gradient(135deg, ${ds.surface}, ${ds.background})`, borderBottom: `1px solid ${ds.primary}30` }}>
   353→        <button onClick={onClose} className="absolute top-4 right-4 w-8 h-8 rounded-full flex items-center justify-center transition-colors" style={{ background: `${ds.primary}20`, color: ds.text }}>
   354→          <X size={16} />
   355→        </button>
   356→        <div className="flex items-start gap-4 pr-10">
   357→          <div className="w-20 h-20 rounded-xl overflow-hidden flex-shrink-0 relative" style={{ border: `1px solid ${ds.primary}40` }}>
   358→            {collection.coverImage && (
   359→              <Image src={collection.coverImage} alt={collection.name} fill className="object-cover" unoptimized />
   360→            )}
   361→          </div>
   362→          <div className="flex-1 min-w-0">
   363→            <div className="flex items-center gap-2 mb-1">
   364→              <p className="text-[10px] tracking-[0.3em] uppercase" style={{ color: ds.primary }}>{collection.family}</p>
   365→              <span className={`text-[9px] px-1.5 py-0.5 rounded-full border ${TIER_STYLES[collection.tier]}`}>{collection.tier}</span>
   366→              <span className="text-[9px] px-1.5 py-0.5 rounded-full" style={{ background: `${ds.primary}20`, color: ds.primary }}>{CATEGORY_LABELS[collection.category]}</span>
   367→            </div>
   368→            <h2 className="font-serif text-3xl font-bold" style={{ color: ds.text, fontFamily: ds.fontDisplay }}>{collection.name}</h2>
   369→            <p className="text-sm mt-1" style={{ color: ds.textMuted }}>{collection.description}</p>
   370→            <div className="flex flex-wrap items-center gap-3 mt-3 text-[11px]" style={{ color: ds.textMuted }}>
   371→              <span className="flex items-center gap-1"><Layers size={11} /> {collection.stats.packs} packs</span>
   372→              <span className="flex items-center gap-1"><Gem size={11} /> {collection.stats.modules} modules</span>
   373→              <span className="flex items-center gap-1"><Sparkles size={11} /> {collection.stats.variants} variantes</span>
   374→              <span className="flex items-center gap-1"><Star size={11} style={{ color: ds.primary }} /> Qualité {collection.stats.qualityScore}%</span>
   375→              <span className="flex items-center gap-1"><Check size={11} style={{ color: ds.primary }} /> {collection.completionPct}% complet</span>
   376→            </div>
   377→          </div>
   378→        </div>
   379→      </div>
   380→
   381→      {/* Pack tabs */}
   382→      <div className="px-6 py-3 flex gap-2 overflow-x-auto custom-scrollbar" style={{ borderBottom: `1px solid ${ds.primary}20`, background: ds.background }}>
   383→        {fullCollection?.packs.map((p) => (
   384→          <PackTabButton key={p.id} pack={p} active={activePack === p.id} onClick={() => setActivePack(p.id)} ds={ds} />
   385→        ))}
   386→      </div>
   387→
   388→      {/* Module previews */}
   389→      <div className="flex-1 overflow-y-auto custom-scrollbar p-6" style={{ background: ds.background }}>
   390→        {pack ? (
   391→          <>
   392→            <p className="text-xs mb-4" style={{ color: ds.textMuted }}>{pack.description} — {pack.modules.length} modules</p>
   393→            <ModulePreview pack={pack} ds={ds} couple={couple} />
   394→          </>
   395→        ) : (
   396→          <div className="flex items-center justify-center py-20">
   397→            <Sparkles size={24} style={{ color: ds.primary }} className="animate-pulse" />
   398→          </div>
   399→        )}
   400→      </div>
   401→
   402→      {/* Footer — Deploy CTA (Phase D — wired to POST /api/collections/deploy) */}
   403→      <div className="p-4 flex flex-col gap-2" style={{ background: ds.surface, borderTop: `1px solid ${ds.primary}30` }}>
   404→        {deployResult && (
   405→          <div
   406→            className="text-xs px-3 py-2 rounded-md flex items-start gap-2"
   407→            style={{
   408→              background: deployResult.ok ? `${ds.primary}15` : 'rgba(239, 68, 68, 0.12)',
   409→              color: deployResult.ok ? ds.primary : '#f87171',
   410→              border: `1px solid ${deployResult.ok ? `${ds.primary}40` : 'rgba(239, 68, 68, 0.3)'}`,
   411→            }}
   412→          >
   413→            {deployResult.ok ? <Check size={14} className="mt-0.5 flex-shrink-0" /> : <X size={14} className="mt-0.5 flex-shrink-0" />}
   414→            <span>{deployResult.message}</span>
   415→          </div>
   416→        )}
   417→        <div className="flex items-center justify-between gap-3">
   418→          <div>
   419→            <p className="text-[10px] uppercase tracking-wider" style={{ color: ds.textMuted }}>Tarif Collection</p>
   420→            <p className="font-serif text-xl font-bold" style={{ color: ds.primary }}>{collection.priceFcfa.toLocaleString('fr-FR')} FCFA</p>
   421→            <p className="text-[10px]" style={{ color: ds.textMuted }}>≈ ${collection.priceUsd} USD</p>
   422→          </div>
   423→          <Button
   424→            onClick={handleDeploy}
   425→            disabled={deploying}
   426→            className="gap-2"
   427→            style={{ background: ds.primary, color: ds.background }}
   428→          >
   429→            {deploying ? <Loader2 size={15} className="animate-spin" /> : <Rocket size={15} />}
   430→            {deploying ? 'Déploiement…' : 'Déployer cette Collection'}
   431→            {!deploying && <ArrowRight size={15} />}
   432→          </Button>
   433→        </div>
   434→      </div>
   435→    </DialogContent>
   436→  )
   437→}
