     3→     3→// ════════════════════════════════════════════════════════════════════════════
     4→     4→// PenpotStudio — Native Penpot Design Studio Integration
     5→     5→// ════════════════════════════════════════════════════════════════════════════
     6→     6→// This is the official design Studio of Wedding OS, powered by Penpot.
     7→     7→//
     8→     8→// Architecture (reuses existing engines, zero regression):
     9→     9→//   ┌─────────────────────────────────────────────────────────────┐
    10→    10→//   │ PenpotStudio (this component)                              │
    11→    11→//   │  • Embeds Penpot editor via iframe (view mode)             │
    12→    12→//   │  • Links a Penpot file URL → stored in Theme.customizations│
    13→    13→//   │  • Pushes theme tokens to Penpot (colors + fonts)          │
    14→    14→//   │  • Pulls design tokens from Penpot                         │
    15→    15→//   │  • Coexists with LuxuryVisualEngine (ambiance overlay)     │
    16→    16→//   ├─────────────────────────────────────────────────────────────┤
    17→    17→//   │ Theme Engine (existing)                                    │
    18→    18→//   │  • /api/theme GET/PUT — reads/writes Theme row             │
    19→    19→//   │  • Theme.customizations JSON stores Penpot integration     │
    20→    20→//   │  • ThemeCustomizer still works (4 canonical fields)        │
    21→    21→//   ├─────────────────────────────────────────────────────────────┤
    22→    22→//   │ ThemeInjector (existing)                                   │
    23→    23→//   │  • Injects --theme-* CSS vars (unchanged)                  │
    24→    24→//   │  • Now ALSO injects --penpot-* CSS vars (additive)         │
    25→    25→//   ├─────────────────────────────────────────────────────────────┤
    26→    26→//   │ InvitationCard (existing, fallback)                        │
    27→    27→//   │  • If Theme.customizations.penpot.invitationFrameId is set │
    28→    28→//   │    → render Penpot SVG export with guest data overlaid     │
    29→    29→//   │  • Else → render the existing fixed InvitationCard design  │
    30→    30→//   ├─────────────────────────────────────────────────────────────┤
    31→    31→//   │ Media Engine (existing)                                    │
    32→    32→//   │  • /api/media returns the wedding's uploaded photos        │
    33→    33→//   │  • Penpot can reference these URLs as placed images        │
    34→    34→//   └─────────────────────────────────────────────────────────────┘
    35→    35→//
    36→    36→// All Penpot state is persisted in Theme.customizations.penpot (JSON).
    37→    37→// This is additive: the existing Theme.customizations field is reused,
    38→    38→// zero schema migration needed.
    39→    39→// ════════════════════════════════════════════════════════════════════════════

    44→    44→import { Input } from '@/components/ui/input'

    46→    46→import { Badge } from '@/components/ui/badge'
    47→    47→import { Separator } from '@/components/ui/separator'

    49→    49→  ExternalLink,

    75→    75→interface PenpotStudioProps {
    76→    76→  /** Wedding slug — used for the X-Wedding-Slug header (tenant scoping). */
    77→    77→  slug?: string
    78→    78→  /**
    79→    79→   * Called when the user links/unlinks a Penpot file.
    80→    80→   * The parent should persist `integration.fileUrl` to Theme.customizations.
    81→    81→   */
    82→    82→  onIntegrationChange?: (integration: PenpotIntegration) => void
    83→    83→}
    84→    84→
    85→    85→export function PenpotStudio({ slug, onIntegrationChange }: PenpotStudioProps) {
    86→    86→  // ─── Load existing Penpot integration from /api/theme ────────────────────
    87→    87→  // The Theme.customizations JSON field stores the PenpotIntegration blob.
    88→    88→  // We read it on mount and expose editors for each field.
    89→    89→  const [loading, setLoading] = useState(true)
    90→    90→  const [saving, setSaving] = useState(false)
    91→    91→  const [syncing, setSyncing] = useState<'push' | 'pull' | null>(null)
    92→    92→  const [integration, setIntegration] = useState<PenpotIntegration>(EMPTY_PENPOT_INTEGRATION)
    93→    93→  const [theme, setTheme] = useState<{
    94→    94→    primaryColor?: string | null
    95→    95→    accentColor?: string | null
    96→    96→    fontDisplay?: string | null
    97→    97→    fontBody?: string | null
    98→    98→  }>({})
    99→    99→  const [fileUrlInput, setFileUrlInput] = useState('')
   100→   100→  const iframeRef = useRef<HTMLIFrameElement>(null)
   101→   101→
   102→   102→  // ─── Fetch current theme + Penpot integration on mount ───────────────────
   103→   103→  const fetchTheme = useCallback(async () => {
   104→   104→    setLoading(true)
   105→   105→    try {
   106→   106→      const res = await fetch('/api/theme')
   107→   107→      if (!res.ok) return
   108→   108→      const data = await res.json()
   109→   109→      const themeRow = data.theme || data
   110→   110→      setTheme({
   111→   111→        primaryColor: themeRow.primaryColor,
   112→   112→        accentColor: themeRow.accentColor,
   113→   113→        fontDisplay: themeRow.fontDisplay,
   114→   114→        fontBody: themeRow.fontBody,
   115→   115→      })
   116→   116→      // Parse the existing customizations JSON (if any) to restore Penpot state
   117→   117→      let customizations: Record<string, unknown> = {}
   118→   118→      if (themeRow.customizations) {
   119→   119→        try {
   120→   120→          customizations =
   121→   121→            typeof themeRow.customizations === 'string'
   122→   122→              ? JSON.parse(themeRow.customizations)
   123→   123→              : themeRow.customizations
   124→   124→        } catch {
   125→   125→          customizations = {}
   126→   126→        }
   127→   127→      }
   128→   128→      const penpot = (customizations.penpot as PenpotIntegration) || EMPTY_PENPOT_INTEGRATION
   129→   129→      setIntegration(penpot)
   130→   130→      setFileUrlInput(penpot.fileUrl || '')
   131→   131→    } catch {
   132→   132→      // Silent — theme may not exist yet for a fresh wedding
   133→   133→    } finally {
   134→   134→      setLoading(false)
   135→   135→    }
   136→   136→  }, [])
   137→   137→
   138→   138→  useEffect(() => {
   139→   139→    fetchTheme()
   140→   140→  }, [fetchTheme])
   141→   141→
   142→   142→  // ─── Persist Penpot integration to /api/theme (PUT) ──────────────────────
   143→   143→  // We merge our PenpotIntegration into the existing Theme.customizations JSON
   144→   144→  // so we never clobber other customizations fields (additive merge).
   145→   145→  // Returns true on success, false on failure (so callers can decide whether
   146→   146→  // to show their own success toast or not).
   147→   147→  const persistIntegration = useCallback(
   148→   148→    async (next: PenpotIntegration): Promise<boolean> => {
   149→   149→      setSaving(true)
   150→   150→      try {
   151→   151→        // First, re-fetch the latest theme to get the current customizations
   152→   152→        // (avoids clobbering concurrent edits from ThemeCustomizer).
   153→   153→        const getRes = await fetch('/api/theme')
   154→   154→        if (!getRes.ok) {
   155→   155→          toast.error('Impossible de charger le thème actuel')
   156→   156→          return false
   157→   157→        }
   158→   158→        const getData = await getRes.json()
   159→   159→        const themeRow = getData.theme || getData
   160→   160→        let customizations: Record<string, unknown> = {}
   161→   161→        if (themeRow.customizations) {
   162→   162→          try {
   163→   163→            customizations =
   164→   164→              typeof themeRow.customizations === 'string'
   165→   165→                ? JSON.parse(themeRow.customizations)
   166→   166→                : themeRow.customizations
   167→   167→          } catch {
   168→   168→            customizations = {}
   169→   169→          }
   170→   170→        }
   171→   171→        // Merge: only touch the `penpot` key, leave everything else intact
   172→   172→        customizations.penpot = next
   173→   173→
   174→   174→        const putRes = await fetch('/api/theme', {
   175→   175→          method: 'PUT',
   176→   176→          headers: { 'Content-Type': 'application/json' },
   177→   177→          body: JSON.stringify({
   178→   178→            // Preserve existing theme fields (don't clobber colors/fonts)
   179→   179→            primaryColor: themeRow.primaryColor,
   180→   180→            accentColor: themeRow.accentColor,
   181→   181→            fontDisplay: themeRow.fontDisplay,
   182→   182→            fontBody: themeRow.fontBody,
   183→   183→            layout: themeRow.layout,
   184→   184→            // Send customizations as an OBJECT — the /api/theme PUT route
   185→   185→            // calls JSON.stringify() itself. Sending a pre-stringified value
   186→   186→            // would cause double-encoding (the route would JSON.stringify a
   187→   187→            // string, producing an escaped string in the DB).
   188→   188→            customizations,
   189→   189→          }),
   190→   190→        })
   191→   191→        if (!putRes.ok) {
   192→   192→          const err = await putRes.json().catch(() => ({}))
   193→   193→          toast.error(err.error || 'Échec de la sauvegarde Penpot')
   194→   194→          return false
   195→   195→        }
   196→   196→        setIntegration(next)
   197→   197→        onIntegrationChange?.(next)
   198→   198→        return true
   199→   199→      } catch {
   200→   200→        toast.error('Erreur réseau lors de la sauvegarde')
   201→   201→        return false
   202→   202→      } finally {
   203→   203→        setSaving(false)
   204→   204→      }
   205→   205→    },
   206→   206→    [onIntegrationChange]
   207→   207→  )
   208→   208→
   209→   209→  // ─── Link a Penpot file ──────────────────────────────────────────────────
   210→   210→  const handleLinkFile = useCallback(async () => {
   211→   211→    const url = fileUrlInput.trim()
   212→   212→    if (!url) {
   213→   213→      toast.error('Collez une URL de fichier Penpot')
   214→   214→      return
   215→   215→    }
   216→   216→    if (!url.startsWith(PENPOT_BASE_URL) && !url.startsWith('http')) {
   217→   217→      toast.error(`L'URL doit commencer par ${PENPOT_BASE_URL}`)
   218→   218→      return
   219→   219→    }
   220→   220→    const { fileId, pageId } = parsePenpotUrl(url)
   221→   221→    if (!fileId) {
   222→   222→      toast.error('URL Penpot invalide — file-id introuvable')
   223→   223→      return
   224→   224→    }
   225→   225→    const next: PenpotIntegration = {
   226→   226→      ...integration,
   227→   227→      fileUrl: url,
   228→   228→      fileId,
   229→   229→      pageId,
   230→   230→      lastSyncedAt: new Date().toISOString(),
   231→   231→    }
   232→   232→    const ok = await persistIntegration(next)
   233→   233→    if (ok) toast.success('Studio Penpot lié avec succès')
   234→   234→  }, [fileUrlInput, integration, persistIntegration])
   235→   235→
   236→   236→  // ─── Unlink Penpot file ──────────────────────────────────────────────────
   237→   237→  const handleUnlink = useCallback(async () => {
   238→   238→    const next: PenpotIntegration = {
   239→   239→      ...EMPTY_PENPOT_INTEGRATION,
   240→   240→      tokens: integration.tokens, // keep tokens even after unlink
   241→   241→    }
   242→   242→    setFileUrlInput('')
   243→   243→    const ok = await persistIntegration(next)
   244→   244→    if (ok) toast.success('Fichier Penpot délié')
   245→   245→  }, [integration, persistIntegration])
   246→   246→
   247→   247→  // ─── Push theme tokens to Penpot ─────────────────────────────────────────
   248→   248→  // Serializes the current ThemeCustomizer colors/fonts as PenpotTokens and
   249→   249→  // stores them in Theme.customizations.penpot.tokens. The couple can then
   250→   250→  // import these tokens into their Penpot file (via Penpot's Tokens plugin
   251→   251→  // or by copying the JSON from the clipboard).
   252→   252→  const handlePushTokens = useCallback(async () => {
   253→   253→    setSyncing('push')
   254→   254→    try {
   255→   255→      const tokens = themeToPenpotTokens(theme)
   256→   256→      const next: PenpotIntegration = {
   257→   257→        ...integration,
   258→   258→        tokens,
   259→   259→        lastSyncedAt: new Date().toISOString(),
   260→   260→      }
   261→   261→      const ok = await persistIntegration(next)
   262→   262→      if (!ok) return
   263→   263→      // Also copy the tokens JSON to clipboard for easy paste into Penpot
   264→   264→      try {
   265→   265→        await navigator.clipboard.writeText(JSON.stringify(tokens, null, 2))
   266→   266→        toast.success('Tokens poussés vers Penpot (JSON copié dans le presse-papiers)')
   267→   267→      } catch {
   268→   268→        toast.success('Tokens poussés vers Penpot')
   269→   269→      }
   270→   270→    } finally {
   271→   271→      setSyncing(null)
   272→   272→    }
   273→   273→  }, [theme, integration, persistIntegration])
   274→   274→
   275→   275→  // ─── Pull tokens from Penpot (manual paste) ──────────────────────────────
   276→   276→  // Since we can't directly read from the Penpot iframe (cross-origin),
   277→   277→  // the couple pastes the Penpot tokens JSON into a prompt. We parse it,
   278→   278→  // store it in customizations, AND update the canonical theme fields
   279→   279→  // (primaryColor, accentColor, fontDisplay, fontBody) so ThemeInjector
   280→   280→  // picks them up immediately.
   281→   281→  const handlePullTokens = useCallback(async () => {
   282→   282→    setSyncing('pull')
   283→   283→    try {
   284→   284→      const pasted = window.prompt(
   285→   285→        'Collez le JSON des tokens exporté depuis Penpot (ou laissez vide pour ignorer) :'
   286→   286→      )
   287→   287→      if (!pasted) {
   288→   288→        setSyncing(null)
   289→   289→        return
   290→   290→      }
   291→   291→      let tokens: PenpotTokens
   292→   292→      try {
   293→   293→        tokens = JSON.parse(pasted)
   294→   294→      } catch {
   295→   295→        toast.error('JSON invalide — vérifiez le format')
   296→   296→        setSyncing(null)
   297→   297→        return
   298→   298→      }
   299→   299→      // Map back to canonical theme fields and persist via /api/theme PUT
   300→   300→      const themeUpdate = penpotTokensToTheme(tokens)
   301→   301→      const getRes = await fetch('/api/theme')
   302→   302→      if (!getRes.ok) {
   303→   303→        toast.error('Impossible de charger le thème')
   304→   304→        return
   305→   305→      }
   306→   306→      const getData = await getRes.json()
   307→   307→      const themeRow = getData.theme || getData
   308→   308→      let customizations: Record<string, unknown> = {}
   309→   309→      if (themeRow.customizations) {
   310→   310→        try {
   311→   311→          customizations =
   312→   312→            typeof themeRow.customizations === 'string'
   313→   313→              ? JSON.parse(themeRow.customizations)
   314→   314→              : themeRow.customizations
   315→   315→        } catch {
   316→   316→          customizations = {}
   317→   317→        }
   318→   318→      }
   319→   319→      const next: PenpotIntegration = {
   320→   320→        ...integration,
   321→   321→        tokens,
   322→   322→        lastSyncedAt: new Date().toISOString(),
   323→   323→      }
   324→   324→      customizations.penpot = next
   325→   325→
   326→   326→      const putRes = await fetch('/api/theme', {
   327→   327→        method: 'PUT',
   328→   328→        headers: { 'Content-Type': 'application/json' },
   329→   329→        body: JSON.stringify({
   330→   330→          primaryColor: themeUpdate.primaryColor ?? themeRow.primaryColor,
   331→   331→          accentColor: themeUpdate.accentColor ?? themeRow.accentColor,
   332→   332→          fontDisplay: themeUpdate.fontDisplay ?? themeRow.fontDisplay,
   333→   333→          fontBody: themeUpdate.fontBody ?? themeRow.fontBody,
   334→   334→          layout: themeRow.layout,
   335→   335→          // Send as object — the /api/theme PUT route does JSON.stringify itself
   336→   336→          customizations,
   337→   337→        }),
   338→   338→      })
   339→   339→      if (!putRes.ok) {
   340→   340→        toast.error('Échec de la synchronisation Penpot → Thème')
   341→   341→        return
   342→   342→      }
   343→   343→      setIntegration(next)
   344→   344→      setTheme((prev) => ({ ...prev, ...themeUpdate }))
   345→   345→      onIntegrationChange?.(next)
   346→   346→      toast.success('Tokens Penpot → Thème synchronisés')
   347→   347→    } finally {
   348→   348→      setSyncing(null)
   349→   349→    }
   350→   350→  }, [integration, persistIntegration, onIntegrationChange])
   351→   351→
   352→   352→  // ─── Render ──────────────────────────────────────────────────────────────
   353→   353→
   354→   354→  if (!PENPOT_ENABLED) {
   355→   355→    return (
   356→   356→      <Card className="border-gold/20">
   357→   357→        <CardContent className="p-6 text-center text-muted-foreground">
   358→   358→          <Palette className="w-8 h-8 mx-auto mb-2 opacity-50" />
   359→   359→          <p>Studio Penpot désactivé. Configurez NEXT_PUBLIC_PENPOT_BASE_URL.</p>
   360→   360→        </CardContent>
   361→   361→      </Card>
   362→   362→    )
   363→   363→  }
   364→   364→
   365→   365→  if (loading) {
   366→   366→    return (
   367→   367→      <div className="flex items-center justify-center py-16">
   368→   368→        <Loader2 className="w-6 h-6 animate-spin text-gold" />
   369→   369→        <span className="ml-2 text-sm text-muted-foreground">Chargement du Studio…</span>
   370→   370→      </div>
   371→   371→    )
   372→   372→  }
   373→   373→
   374→   374→  const isLinked = !!(integration.fileId && integration.fileUrl)
   375→   375→  const viewUrl = isLinked
   376→   376→    ? buildPenpotViewUrl(integration.fileId!, integration.pageId)
   377→   377→    : null
   378→   378→  const editUrl = isLinked
   379→   379→    ? buildPenpotEditUrl(integration.fileId!, integration.pageId)
   380→   380→    : null
   381→   381→
   382→   382→  return (
   383→   383→    <div className="space-y-4 p-4 md:p-6">
   384→   384→      {/* ─── Header ─────────────────────────────────────────────────────── */}
   385→   385→      <Card className="border-gold/20 bg-gradient-to-br from-gold/[0.03] to-transparent">
   386→   386→        <CardHeader className="pb-3">
   387→   387→          <div className="flex items-start justify-between gap-3">
   388→   388→            <div className="flex items-center gap-3">
   389→   389→              <div className="w-10 h-10 rounded-lg bg-gradient-to-br from-gold/20 to-rose-gold/10 flex items-center justify-center">
   390→   390→                <Palette className="w-5 h-5 text-gold" />
   391→   391→              </div>
   392→   392→              <div>
   393→   393→                <CardTitle className="text-lg flex items-center gap-2">
   394→   394→                  Studio Penpot
   395→   395→                  {isLinked ? (
   396→   396→                    <Badge variant="outline" className="text-emerald-400 border-emerald-400/30 text-xs">
   397→   397→                      <CheckCircle2 className="w-3 h-3 mr-1" />
   398→   398→                      Lié
   399→   399→                    </Badge>
   400→   400→                  ) : (
   401→   401→                    <Badge variant="outline" className="text-muted-foreground text-xs">
   402→   402→                      Non lié
   403→   403→                    </Badge>
   404→   404→                  )}
   405→   405→                </CardTitle>
   406→   406→                <p className="text-xs text-muted-foreground mt-1">
   407→   407→                  Le Studio officiel de création — invitations, save-the-date, visuels
   408→   408→                </p>
   409→   409→              </div>
   410→   410→            </div>
   411→   411→            {isLinked && editUrl && (
   412→   412→              <Button asChild size="sm" variant="outline" className="shrink-0">
   413→   413→                <a href={editUrl} target="_blank" rel="noopener noreferrer">
   414→   414→                  <ExternalLink className="w-3.5 h-3.5 mr-1.5" />
   415→   415→                  Éditer dans Penpot
   416→   416→                </a>
   417→   417→              </Button>
   418→   418→            )}
   419→   419→          </div>
   420→   420→        </CardHeader>
   421→   421→      </Card>
   422→   422→
   423→   423→      {/* ─── File Linker ────────────────────────────────────────────────── */}
   424→   424→      <Card className="border-white/10">
   425→   425→        <CardHeader className="pb-3">
   426→   426→          <CardTitle className="text-sm flex items-center gap-2">
   427→   427→            <Link2 className="w-4 h-4 text-gold" />
   428→   428→            Fichier Penpot
   429→   429→          </CardTitle>
   430→   430→        </CardHeader>
   431→   431→        <CardContent className="space-y-3">
   432→   432→          <div className="space-y-2">
   433→   433→            <Label htmlFor="penpot-url" className="text-xs">
   434→   434→              URL du fichier Penpot (partage ou éditeur)
   435→   435→            </Label>
   436→   436→            <div className="flex gap-2">
   437→   437→              <Input
   438→   438→                id="penpot-url"
   439→   439→                value={fileUrlInput}
   440→   440→                onChange={(e) => setFileUrlInput(e.target.value)}
   441→   441→                placeholder={`${PENPOT_BASE_URL}/#/view?file-id=...&page-id=...`}
   442→   442→                className="text-xs font-mono"
   443→   443→                disabled={saving}
   444→   444→              />
   445→   445→              {isLinked ? (
   446→   446→                <Button
   447→   447→                  variant="outline"
   448→   448→                  size="sm"
   449→   449→                  onClick={handleUnlink}
   450→   450→                  disabled={saving}
