     4→function generateCode(): string {
     5→  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'
     6→  let code = ''
     7→  for (let i = 0; i < 8; i++) {
     8→    code += chars.charAt(Math.floor(Math.random() * chars.length))
     9→  }
    10→  return code
    11→}
