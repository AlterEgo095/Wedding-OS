   236→
   237→function formatUsd(cents: number): string {
   238→  return `$${(cents / 100).toFixed(2)}`
   239→}
   240→
   241→function formatFcfa(cents: number): string {
   242→  const usd = cents / 100
   243→  return `${Math.round(usd * 600).toLocaleString('fr-FR')} FCFA`
   244→}
   245→
   246→function formatDate(iso: string | null): string {
   247→  if (!iso) return '—'
   248→  return new Date(iso).toLocaleDateString('fr-FR', {
   249→    day: '2-digit',
   250→    month: 'short',
   251→    year: 'numeric',
   252→    hour: '2-digit',
   253→    minute: '2-digit',
   254→  })
   255→}

   258→// Main BillingTab component
   259→// ══════════════════════════════════════════════════════════════════════════════
   260→
   261→export function BillingTab({
   262→  fetchWithAuth,
   263→}: {
   264→  fetchWithAuth: (url: string, init?: RequestInit) => Promise<Response | null>
   265→}) {
   266→  const [data, setData] = useState<BillingOverviewResponse | null>(null)
   267→  const [loading, setLoading] = useState(true)
   268→  const [search, setSearch] = useState('')
   269→  const [statusFilter, setStatusFilter] = useState('ALL')
   270→  const [planFilter, setPlanFilter] = useState('ALL')
   271→
   272→  // Editor dialog state
   273→  const [editing, setEditing] = useState<BillingWeddingRow | null>(null)
   274→  const [editorOpen, setEditorOpen] = useState(false)
   275→  const [subscriptionDetail, setSubscriptionDetail] = useState<SubscriptionSummary | null>(null)
   276→  const [invoices, setInvoices] = useState<InvoiceRow[]>([])
   277→  const [form, setForm] = useState<SubscriptionFormState>(EMPTY_FORM)
   278→  const [saving, setSaving] = useState(false)
   279→  const [creatingInvoice, setCreatingInvoice] = useState(false)
   280→  const [actionLoading, setActionLoading] = useState<string | null>(null) // invoice id being acted upon
   281→
   282→  // WhatsApp modal state
   283→  const [whatsappData, setWhatsappData] = useState<WhatsAppResponse | null>(null)
   284→  const [whatsappOpen, setWhatsappOpen] = useState(false)
   285→  const [generatingWhatsapp, setGeneratingWhatsapp] = useState(false)
   286→
   287→  const searchRef = useRef(search)
   288→  searchRef.current = search
   289→
   290→  const load = useCallback(async () => {
   291→    setLoading(true)
   292→    const params = new URLSearchParams()
   293→    if (searchRef.current) params.set('search', searchRef.current)
   294→    if (statusFilter !== 'ALL') params.set('status', statusFilter)
   295→    if (planFilter !== 'ALL') params.set('plan', planFilter)
   296→    const res = await fetchWithAuth(`/api/platform/billing/weddings?${params.toString()}`)
   297→    if (!res) {
   298→      setLoading(false)
   299→      return
   300→    }
   301→    try {
   302→      const json = (await res.json()) as BillingOverviewResponse
   303→      setData(json)
   304→    } catch {
   305→      toast.error('Réponse invalide du serveur')
   306→    } finally {
   307→      setLoading(false)
   308→    }
   309→  }, [fetchWithAuth, statusFilter, planFilter])
   310→
   311→  useEffect(() => {
   312→    load()
   313→  }, [statusFilter, planFilter, load])
   314→
   315→  // Debounced search
   316→  useEffect(() => {
   317→    const t = setTimeout(() => load(), 350)
   318→    return () => clearTimeout(t)
   319→  }, [search, load])
   320→
   321→  // ─── Open subscription editor for a wedding ─────────────────────────────
   322→  const openEditor = async (row: BillingWeddingRow) => {
   323→    setEditing(row)
   324→    setEditorOpen(true)
   325→    setForm(EMPTY_FORM)
   326→    setSubscriptionDetail(null)
   327→    setInvoices([])
   328→
   329→    // Fetch current subscription detail + invoices in parallel
   330→    const [subRes, invRes] = await Promise.all([
   331→      fetchWithAuth(`/api/platform/weddings/${row.id}/subscription`),
   332→      fetchWithAuth(`/api/platform/weddings/${row.id}/invoices`),
   333→    ])
   334→    if (subRes) {
   335→      try {
   336→        const json = (await subRes.json()) as SubscriptionDetailResponse
   337→        if (json.subscription) {
   338→          setSubscriptionDetail(json.subscription)
   339→          setForm({
   340→            plan: json.subscription.plan,
   341→            status: json.subscription.status,
   342→            billingCycle: json.subscription.billingCycle,
   343→            amountAgreed:
   344→              json.subscription.amountAgreed != null
   345→                ? String(json.subscription.amountAgreed)
   346→                : '',
   347→            paymentMethod: json.subscription.paymentMethod || '',
   348→            whatsappPhone: json.subscription.whatsappPhone || '',
   349→            notes: json.subscription.notes || '',
   350→          })
   351→        } else {
   352→          // No subscription yet — default to wedding's current plan
   353→          setForm({
   354→            ...EMPTY_FORM,
   355→            plan: row.plan,
   356→          })
   357→        }
   358→      } catch {
   359→        /* ignore */
   360→      }
   361→    }
   362→    if (invRes) {
   363→      try {
   364→        const json = (await invRes.json()) as InvoicesListResponse
   365→        setInvoices(json.invoices || [])
   366→      } catch {
   367→        /* ignore */
   368→      }
   369→    }
   370→  }
   371→
   372→  // ─── Save subscription ──────────────────────────────────────────────────
   373→  const handleSaveSubscription = async () => {
   374→    if (!editing) return
   375→    setSaving(true)
   376→    const payload: Record<string, unknown> = {
   377→      plan: form.plan,
   378→      status: form.status,
   379→      billingCycle: form.billingCycle,
   380→      paymentMethod: form.paymentMethod || null,
   381→      whatsappPhone: form.whatsappPhone.trim() || null,
   382→      notes: form.notes.trim() || null,
   383→    }
   384→    if (form.amountAgreed.trim()) {
   385→      const n = Number(form.amountAgreed)
   386→      if (!Number.isFinite(n) || n < 0) {
   387→        toast.error('Prix personnalisé invalide (entier en centimes USD)')
   388→        setSaving(false)
   389→        return
   390→      }
   391→      payload.amountAgreed = Math.round(n)
   392→    } else {
   393→      payload.amountAgreed = null
   394→    }
   395→
   396→    const res = await fetchWithAuth(
   397→      `/api/platform/weddings/${editing.id}/subscription`,
   398→      {
   399→        method: 'PUT',
   400→        headers: { 'Content-Type': 'application/json' },
   401→        body: JSON.stringify(payload),
   402→      },
   403→    )
   404→    setSaving(false)
   405→    if (!res) return
   406→    if (res.ok) {
   407→      toast.success('Abonnement enregistré')
   408→      const json = (await res.json()) as { subscription: SubscriptionSummary }
   409→      setSubscriptionDetail(json.subscription)
   410→      await load() // refresh overview
   411→    } else {
   412→      const json = await res.json().catch(() => ({}))
   413→      toast.error(json.error || 'Erreur lors de la sauvegarde')
   414→    }
   415→  }
   416→
   417→  // ─── Generate WhatsApp message ──────────────────────────────────────────
   418→  const handleGenerateWhatsApp = async () => {
   419→    if (!editing) return
   420→    setGeneratingWhatsapp(true)
   421→    const res = await fetchWithAuth(
   422→      `/api/platform/weddings/${editing.id}/subscription/whatsapp`,
   423→      {
   424→        method: 'POST',
   425→        headers: { 'Content-Type': 'application/json' },
   426→        body: JSON.stringify({
   427→          plan: form.plan,
   428→          billingCycle: form.billingCycle,
   429→          whatsappPhone: form.whatsappPhone.trim() || null,
   430→          notes: form.notes.trim() || null,
   431→        }),
   432→      },
   433→    )
   434→    setGeneratingWhatsapp(false)
   435→    if (!res) return
   436→    if (res.ok) {
   437→      const json = (await res.json()) as WhatsAppResponse
   438→      setWhatsappData(json)
   439→      setWhatsappOpen(true)
   440→    } else {
   441→      const json = await res.json().catch(() => ({}))
   442→      toast.error(json.error || 'Erreur lors de la génération du message')
   443→    }
   444→  }
   445→
   446→  // ─── Create invoice ─────────────────────────────────────────────────────
   447→  const handleCreateInvoice = async () => {
   448→    if (!editing) return
   449→    setCreatingInvoice(true)
   450→    const res = await fetchWithAuth(
   451→      `/api/platform/weddings/${editing.id}/invoices`,
   452→      {
   453→        method: 'POST',
   454→        headers: { 'Content-Type': 'application/json' },
   455→        body: JSON.stringify({
   456→          plan: form.plan,
   457→          billingCycle: form.billingCycle,
   458→          paymentMethod: form.paymentMethod || null,
   459→          whatsappPhone: form.whatsappPhone.trim() || null,
   460→          notes: form.notes.trim() || null,
   461→        }),
   462→      },
   463→    )
   464→    setCreatingInvoice(false)
   465→    if (!res) return
   466→    if (res.ok) {
   467→      toast.success('Facture créée')
   468→      const json = (await res.json()) as { invoice: InvoiceRow }
   469→      setInvoices((prev) => [json.invoice, ...prev])
   470→      await load()
   471→    } else {
   472→      const json = await res.json().catch(() => ({}))
   473→      toast.error(json.error || 'Erreur lors de la création de la facture')
   474→    }
   475→  }
   476→
   477→  // ─── Mark invoice as paid ───────────────────────────────────────────────
   478→  const handleMarkPaid = async (invoice: InvoiceRow) => {
   479→    if (!editing) return
   480→    setActionLoading(invoice.id)
   481→    const res = await fetchWithAuth(`/api/platform/invoices/${invoice.id}`, {
   482→      method: 'PUT',
   483→      headers: { 'Content-Type': 'application/json' },
   484→      body: JSON.stringify({
   485→        status: 'PAID',
   486→        paymentMethod: invoice.paymentMethod || form.paymentMethod || null,
   487→      }),
   488→    })
   489→    setActionLoading(null)
   490→    if (!res) return
   491→    if (res.ok) {
   492→      toast.success('Facture marquée comme payée')
   493→      const json = (await res.json()) as { invoice: InvoiceRow }
   494→      setInvoices((prev) =>
   495→        prev.map((inv) => (inv.id === invoice.id ? json.invoice : inv)),
   496→      )
   497→      await load()
   498→    } else {
   499→      const json = await res.json().catch(() => ({}))
   500→      toast.error(json.error || 'Erreur lors du marquage')
   501→    }
   502→  }
   503→
   504→  // ─── Void invoice ───────────────────────────────────────────────────────
   505→  const handleVoidInvoice = async (invoice: InvoiceRow) => {
   506→    if (!editing) return
   507→    setActionLoading(invoice.id)
   508→    const res = await fetchWithAuth(`/api/platform/invoices/${invoice.id}`, {
   509→      method: 'PUT',
   510→      headers: { 'Content-Type': 'application/json' },
   511→      body: JSON.stringify({ status: 'VOID' }),
   512→    })
   513→    setActionLoading(null)
   514→    if (!res) return
   515→    if (res.ok) {
   516→      toast.success('Facture annulée')
   517→      const json = (await res.json()) as { invoice: InvoiceRow }
   518→      setInvoices((prev) =>
   519→        prev.map((inv) => (inv.id === invoice.id ? json.invoice : inv)),
   520→      )
   521→      await load()
   522→    } else {
   523→      const json = await res.json().catch(() => ({}))
   524→      toast.error(json.error || 'Erreur lors de l\'annulation')
   525→    }
   526→  }
   527→
   528→  // ─── Copy WhatsApp message to clipboard ─────────────────────────────────
   529→  const handleCopyMessage = async () => {
   530→    if (!whatsappData) return
   531→    try {
   532→      await navigator.clipboard.writeText(whatsappData.message)
   533→      toast.success('Message copié dans le presse-papiers')
   534→    } catch {
   535→      toast.error('Impossible de copier le message')
   536→    }
   537→  }
   538→
   539→  // ════════════════════════════════════════════════════════════════════════════
   540→  // Render
   541→  // ════════════════════════════════════════════════════════════════════════════
   542→
   543→  const summary = data?.summary
   544→
   545→  return (
   546→    <div className="space-y-6">
   547→      {/* ─── Summary cards ─────────────────────────────────────────────── */}
   548→      <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-5 gap-3">
   549→        <SummaryCard
   550→          icon={<Wallet className="w-4 h-4" />}
   551→          label="Total mariages"
   552→          value={summary?.total ?? 0}
   553→          loading={loading}
   554→        />
   555→        <SummaryCard
   556→          icon={<CheckCircle2 className="w-4 h-4" />}
   557→          label="Actifs"
   558→          value={summary?.active ?? 0}
   559→          loading={loading}
   560→          accent="emerald"
   561→        />
   562→        <SummaryCard
   563→          icon={<Clock className="w-4 h-4" />}
   564→          label="En attente"
   565→          value={summary?.pending ?? 0}
   566→          loading={loading}
   567→          accent="amber"
   568→        />
   569→        <SummaryCard
   570→          icon={<TrendingUp className="w-4 h-4" />}
   571→          label="MRR (USD)"
   572→          value={`$${(summary?.mrrUsd ?? 0).toFixed(2)}`}
   573→          loading={loading}
   574→          accent="gold"
   575→        />
   576→        <SummaryCard
   577→          icon={<DollarSign className="w-4 h-4" />}
   578→          label="À recouvrer"
   579→          value={`$${(summary?.pendingUsd ?? 0).toFixed(2)}`}
   580→          loading={loading}
   581→          accent="orange"
   582→        />
   583→      </div>
   584→
   585→      {/* ─── Filters ────────────────────────────────────────────────────── */}
