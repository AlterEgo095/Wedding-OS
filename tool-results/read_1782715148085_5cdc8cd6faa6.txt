   275→// usePlatformFetch — wraps fetch with Bearer token + session-expiry handling
   276→// ════════════════════════════════════════════════════════════════════════════
   277→
   278→function usePlatformFetch() {
   279→  const router = useRouter()
   280→  const sessionExpiredRef = useRef(false)
   281→
   282→  const onSessionExpired = useCallback(() => {
   283→    if (sessionExpiredRef.current) return
   284→    sessionExpiredRef.current = true
   285→    try {
   286→      localStorage.removeItem('admin_token')
   287→      localStorage.removeItem('admin_user')
   288→    } catch {
   289→      /* ignore */
   290→    }
   291→    toast.error('Session expirée, veuillez vous reconnecter')
   292→    router.replace('/platform/login')
   293→  }, [router])
   294→
   295→  const getToken = useCallback(() => {
   296→    if (typeof window === 'undefined') return null
   297→    return localStorage.getItem('admin_token')
   298→  }, [])
   299→
   300→  const fetchWithAuth = useCallback(
   301→    async (url: string, init?: RequestInit): Promise<Response | null> => {
   302→      const token = getToken()
   303→      if (!token) {
   304→        onSessionExpired()
   305→        return null
   306→      }
   307→      let res: Response
   308→      try {
   309→        res = await fetch(url, {
   310→          ...init,
   311→          headers: {
   312→            ...(init?.headers || {}),
   313→            Authorization: `Bearer ${token}`,
   314→          },
   315→        })
   316→      } catch {
   317→        toast.error('Erreur de connexion au serveur')
   318→        return null
   319→      }
   320→      if (res.status === 401) {
   321→        onSessionExpired()
   322→        return null
   323→      }
   324→      if (res.status === 403) {
   325→        toast.error('Accès refusé')
   326→        return null
   327→      }
   328→      return res
   329→    },
   330→    [getToken, onSessionExpired]
   331→  )
   332→
   333→  return { fetchWithAuth, onSessionExpired, getToken }
   334→}

   364→function formatDate(iso: string | null): string {
   365→  if (!iso) return '—'
   366→  try {
   367→    return new Date(iso).toLocaleDateString('fr-FR', {
   368→      day: '2-digit',
   369→      month: 'short',
   370→      year: 'numeric',
   371→    })
   372→  } catch {
   373→    return '—'
   374→  }
   375→}
   376→
   377→function formatDateTime(iso: string | null): string {
   378→  if (!iso) return '—'
   379→  try {
   380→    return new Date(iso).toLocaleString('fr-FR', {
   381→      day: '2-digit',
   382→      month: '2-digit',
   383→      year: 'numeric',
   384→      hour: '2-digit',
   385→      minute: '2-digit',
   386→    })
   387→  } catch {
   388→    return '—'
   389→  }
   390→}

   405→function DashboardTab({ fetchWithAuth }: { fetchWithAuth: (url: string, init?: RequestInit) => Promise<Response | null> }) {
   406→  const [data, setData] = useState<DashboardData | null>(null)
   407→  const [loading, setLoading] = useState(true)
   408→
   409→  const load = useCallback(async () => {
   410→    setLoading(true)
   411→    const res = await fetchWithAuth('/api/platform/dashboard')
   412→    if (!res) {
   413→      setLoading(false)
   414→      return
   415→    }
   416→    try {
   417→      const json = (await res.json()) as DashboardData
   418→      setData(json)
   419→    } catch {
   420→      toast.error('Réponse invalide du serveur')
   421→    } finally {
   422→      setLoading(false)
   423→    }
   424→  }, [fetchWithAuth])
   425→
   426→  useEffect(() => {
   427→    load()
   428→  }, [load])
   429→
   430→  if (loading) {
   431→    return (
   432→      <div className="space-y-6 p-4 md:p-6">
   433→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   434→          {Array.from({ length: 4 }).map((_, i) => (
   435→            <Skeleton key={i} className="h-28 rounded-xl" />
   436→          ))}
   437→        </div>
   438→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   439→          <Skeleton className="h-96 rounded-xl" />
   440→          <Skeleton className="h-96 rounded-xl" />

   442→      </div>

   773→function WeddingsTab({ fetchWithAuth }: { fetchWithAuth: (url: string, init?: RequestInit) => Promise<Response | null> }) {
   774→  const [weddings, setWeddings] = useState<Wedding[]>([])
   775→  const [total, setTotal] = useState(0)
   776→  const [totalPages, setTotalPages] = useState(1)
   777→  const [page, setPage] = useState(1)
   778→  const [loading, setLoading] = useState(true)
   779→  const [search, setSearch] = useState('')
   780→  const [statusFilter, setStatusFilter] = useState<string>('ALL')
   781→  const [planFilter, setPlanFilter] = useState<string>('ALL')
   782→
   783→  // Dialog state
   784→  const [showFormDialog, setShowFormDialog] = useState(false)
   785→  const [showDeleteDialog, setShowDeleteDialog] = useState(false)
   786→  const [editing, setEditing] = useState<Wedding | null>(null)
   787→  const [deleting, setDeleting] = useState<Wedding | null>(null)
   788→  const [form, setForm] = useState<WeddingFormState>(EMPTY_WEDDING_FORM)
   789→  const [saving, setSaving] = useState(false)
   790→
   791→  const LIMIT = 20
   792→  const searchRef = useRef(search)
   793→  searchRef.current = search
   794→
   795→  const load = useCallback(
   796→    async (targetPage: number) => {
   797→      setLoading(true)
   798→      const params = new URLSearchParams({
   799→        page: String(targetPage),
   800→        limit: String(LIMIT),
   801→      })
   802→      if (searchRef.current) params.set('search', searchRef.current)
   803→      if (statusFilter !== 'ALL') params.set('status', statusFilter)
   804→      if (planFilter !== 'ALL') params.set('plan', planFilter)
   805→
   806→      const res = await fetchWithAuth(`/api/platform/weddings?${params.toString()}`)
   807→      if (!res) {
   808→        setLoading(false)
   809→        return
   810→      }
   811→      try {
   812→        const json = (await res.json()) as PaginatedWeddings
   813→        setWeddings(json.weddings || [])
   814→        setTotal(json.total || 0)
   815→        // API returns { total, page, limit } — compute totalPages client-side.
   816→        setTotalPages(Math.max(1, Math.ceil((json.total || 0) / (json.limit || LIMIT))))
   817→        setPage(json.page || targetPage)

   819→        toast.error('Réponse invalide du serveur')

   821→        setLoading(false)

   824→    [fetchWithAuth, statusFilter, planFilter]

   827→  useEffect(() => {
   828→    load(1)
   829→  }, [statusFilter, planFilter, load])
   830→
   831→  // Debounced search trigger
   832→  useEffect(() => {
   833→    const t = setTimeout(() => {
   834→      load(1)
   835→    }, 350)
   836→    return () => clearTimeout(t)
   837→  }, [search, load])

   840→    setEditing(null)
   841→    setForm(EMPTY_WEDDING_FORM)
   842→    setShowFormDialog(true)
   843→  }

   859→  const handleSave = async () => {

   861→      toast.error('Slug, mariée et marié sont requis')
   862→      return
   863→    }
   864→    setSaving(true)
   865→    const payload: Record<string, unknown> = {
   866→      slug: form.slug.trim().toLowerCase(),
   867→      brideName: form.brideName.trim(),
   868→      groomName: form.groomName.trim(),
   869→      weddingDate: form.weddingDate ? new Date(form.weddingDate).toISOString() : null,
   870→      venueCity: form.venueCity.trim() || null,
   871→      status: form.status,
   872→      plan: form.plan,

   876→        ? `/api/platform/weddings/${editing.id}`
   877→        : '/api/platform/weddings'

   879→      const res = await fetchWithAuth(url, {
   880→        method,
   881→        headers: { 'Content-Type': 'application/json' },
   882→        body: JSON.stringify(payload),
   883→      })
   884→      if (!res) {
   885→        setSaving(false)
   886→        return
   887→      }
   888→      const json = await res.json().catch(() => ({}))

   891→        setShowFormDialog(false)
   892→        setEditing(null)

   920→        setDeleting(null)

   922→      } else {
   923→        toast.error(json.error || 'Erreur lors de la suppression')
   924→      }
   925→    } catch {
   926→      toast.error('Erreur de connexion')
   927→    } finally {
   928→      setSaving(false)
   929→    }
