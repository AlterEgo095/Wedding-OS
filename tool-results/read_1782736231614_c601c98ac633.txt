   176→   176→function formatFcfa(n: number): string {
   177→   177→  if (n === 0) return 'Gratuit'
   178→   178→  return `${n.toLocaleString('fr-FR')} FCFA`
   179→   179→}

   182→   182→  if (n === 0) return ''
   183→   183→  return `$${n}`
