    46→/**
    47→ * Detect the current wedding slug from the URL path.
    48→ * Returns null on root `/admin` (legacy SPA served at root, default wedding).
    49→ * Returns the slug on `/w/[slug]/...` routes (per-wedding admin).
    50→ */
    51→function getWeddingSlug(): string | null {
    52→  if (typeof window === 'undefined') return null
    53→  const match = window.location.pathname.match(/^\/w\/([a-z0-9-]+)/i)
    54→  return match?.[1] ?? null
    55→}
    56→
    57→/**
    58→ * Build the public invite URL for a guest. On the legacy root `/admin` SPA this
    59→ * points to `/?invite=…` (default wedding). On per-wedding admin it points to
    60→ * `/w/<slug>?invite=…` so the invitee lands on the correct tenant site.
    61→ */
    62→function buildInviteUrl(suffix: string): string {
    63→  if (typeof window === 'undefined') return ''
    64→  const origin = window.location.origin
    65→  const slug = getWeddingSlug()
    66→  return slug
    67→    ? `${origin}/w/${slug}${suffix}`
    68→    : `${origin}${suffix}`
    69→}
