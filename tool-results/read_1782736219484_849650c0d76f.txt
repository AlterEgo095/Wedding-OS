   176→function formatFcfa(n: number): string {
   177→  if (n === 0) return 'Gratuit'
   178→  return `${n.toLocaleString('fr-FR')} FCFA`
   179→}
   180→
   181→function formatUsd(n: number): string {
   182→  if (n === 0) return ''
   183→  return `$${n}`
