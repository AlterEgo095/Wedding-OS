    31→/* ─── Map activity keywords to icon name string ─── */
    32→function getEventIconName(activity: string): string {
    33→  const lower = activity.toLowerCase()
    34→  if (lower.includes('église') || lower.includes('cérémonie') || lower.includes('church') || lower.includes('mariage'))
    35→    return 'church'
    36→  if (lower.includes('musique') || lower.includes('dance') || lower.includes('danse') || lower.includes('music') || lower.includes('dj') || lower.includes('soirée'))
    37→    return 'music'
    38→  if (lower.includes('repas') || lower.includes('dîner') || lower.includes('dinner') || lower.includes('buffet') || lower.includes('cocktail') || lower.includes('réception'))
    39→    return 'utensils'
    40→  if (lower.includes('gâteau') || lower.includes('cake') || lower.includes('dessert'))
    41→    return 'cake'
    42→  if (lower.includes('photo') || lower.includes('préparatifs') || lower.includes('séance') || lower.includes('camera'))
    43→    return 'camera'
    44→  if (lower.includes('champagne') || lower.includes('toast') || lower.includes('vin') || lower.includes('accueil'))
    45→    return 'wine'
    46→  if (lower.includes('vœux') || lower.includes('coeur') || lower.includes('amour'))
    47→    return 'heart'
    48→  if (lower.includes('feu') || lower.includes('artifice') || lower.includes('sparkle') || lower.includes('entrée'))
    49→    return 'sparkles'
    50→  return 'wine'
    51→}

    71→/* ─── Floating sparkle particles ─── */
    72→function FloatingParticles() {
    73→  const particles = useMemo(
    74→    () =>
    75→      Array.from({ length: 14 }, (_, i) => ({
    76→        id: i,
    77→        x: Math.random() * 100,
    78→        y: Math.random() * 100,
    79→        size: Math.random() * 2.5 + 1,
    80→        duration: Math.random() * 7 + 8,
    81→        delay: Math.random() * 5,
    82→        opacity: Math.random() * 0.3 + 0.08,
    83→      })),
    84→    [],
    85→  )
    86→
    87→  return (
    88→    <div className="absolute inset-0 pointer-events-none overflow-hidden z-0" aria-hidden="true">
    89→      {particles.map((p) => (
    90→        <motion.div
    91→          key={p.id}
    92→          className="absolute rounded-full"
    93→          style={{
    94→            left: `${p.x}%`,
    95→            top: `${p.y}%`,
    96→            width: p.size,
    97→            height: p.size,
    98→            background: `radial-gradient(circle, ${GOLD} 0%, transparent 70%)`,
    99→          }}
   100→          animate={{
   101→            y: [0, -25, 0],
   102→            opacity: [p.opacity, p.opacity * 1.5, p.opacity],
   103→            scale: [1, 1.2, 1],
   104→          }}
   105→          transition={{
   106→            duration: p.duration,
   107→            delay: p.delay,
   108→            repeat: Infinity,
   109→            ease: 'easeInOut',
   110→          }}
   111→        />
   112→      ))}
   113→    </div>
   114→  )
   115→}
