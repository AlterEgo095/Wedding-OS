   283→   283→// usePlatformFetch — wraps fetch with Bearer token + session-expiry handling
   284→   284→// ════════════════════════════════════════════════════════════════════════════
   285→   285→
   286→   286→function usePlatformFetch() {
   287→   287→  const router = useRouter()
   288→   288→  const sessionExpiredRef = useRef(false)
   289→   289→
   290→   290→  const onSessionExpired = useCallback(() => {
   291→   291→    if (sessionExpiredRef.current) return
   292→   292→    sessionExpiredRef.current = true
   293→   293→    try {
   294→   294→      localStorage.removeItem('admin_token')
   295→   295→      localStorage.removeItem('admin_user')
   296→   296→    } catch {
   297→   297→      /* ignore */
   298→   298→    }
   299→   299→    toast.error('Session expirée, veuillez vous reconnecter')
   300→   300→    router.replace('/platform/login')
   301→   301→  }, [router])
   302→   302→
   303→   303→  const getToken = useCallback(() => {
   304→   304→    if (typeof window === 'undefined') return null
   305→   305→    return localStorage.getItem('admin_token')
   306→   306→  }, [])
   307→   307→
   308→   308→  const fetchWithAuth = useCallback(
   309→   309→    async (url: string, init?: RequestInit): Promise<Response | null> => {
   310→   310→      const token = getToken()
   311→   311→      if (!token) {
   312→   312→        onSessionExpired()
   313→   313→        return null
   314→   314→      }
   315→   315→      let res: Response
   316→   316→      try {
   317→   317→        res = await fetch(url, {
   318→   318→          ...init,
   319→   319→          headers: {
   320→   320→            ...(init?.headers || {}),
   321→   321→            Authorization: `Bearer ${token}`,
   322→   322→          },
   323→   323→        })
   324→   324→      } catch {
   325→   325→        toast.error('Erreur de connexion au serveur')
   326→   326→        return null
   327→   327→      }
   328→   328→      if (res.status === 401) {
   329→   329→        onSessionExpired()
   330→   330→        return null
   331→   331→      }
   332→   332→      if (res.status === 403) {
   333→   333→        toast.error('Accès refusé')
   334→   334→        return null
   335→   335→      }
   336→   336→      return res
   337→   337→    },
   338→   338→    [getToken, onSessionExpired]
   339→   339→  )
   340→   340→
   341→   341→  return { fetchWithAuth, onSessionExpired, getToken }
   342→   342→}
