   265→// ════════════════════════════════════════════════════════════════════════════
   266→// usePlatformFetch — wraps fetch with Bearer token + session-expiry handling
   267→// ════════════════════════════════════════════════════════════════════════════
   268→
   269→function usePlatformFetch() {
   270→  const router = useRouter()
   271→  const sessionExpiredRef = useRef(false)
   272→
   273→  const onSessionExpired = useCallback(() => {
   274→    if (sessionExpiredRef.current) return
   275→    sessionExpiredRef.current = true
   276→    try {
   277→      localStorage.removeItem('admin_token')
   278→      localStorage.removeItem('admin_user')
   279→    } catch {
   280→      /* ignore */
   281→    }
   282→    toast.error('Session expirée, veuillez vous reconnecter')
   283→    router.replace('/platform/login')
   284→  }, [router])
   285→
   286→  const getToken = useCallback(() => {
   287→    if (typeof window === 'undefined') return null
   288→    return localStorage.getItem('admin_token')
   289→  }, [])
   290→
   291→  const fetchWithAuth = useCallback(
   292→    async (url: string, init?: RequestInit): Promise<Response | null> => {
   293→      const token = getToken()
   294→      if (!token) {
   295→        onSessionExpired()
   296→        return null
   297→      }
   298→      let res: Response
   299→      try {
   300→        res = await fetch(url, {
   301→          ...init,
   302→          headers: {
   303→            ...(init?.headers || {}),
   304→            Authorization: `Bearer ${token}`,
   305→          },
   306→        })
   307→      } catch {
   308→        toast.error('Erreur de connexion au serveur')
   309→        return null
   310→      }
   311→      if (res.status === 401) {
   312→        onSessionExpired()
   313→        return null
   314→      }
   315→      if (res.status === 403) {
   316→        toast.error('Accès refusé')
   317→        return null
   318→      }
   319→      return res
   320→    },
   321→    [getToken, onSessionExpired]
   322→  )
   323→
   324→  return { fetchWithAuth, onSessionExpired, getToken }
   325→}

   355→function formatDate(iso: string | null): string {
   356→  if (!iso) return '—'
   357→  try {
   358→    return new Date(iso).toLocaleDateString('fr-FR', {
   359→      day: '2-digit',
   360→      month: 'short',
   361→      year: 'numeric',
   362→    })
   363→  } catch {
   364→    return '—'
   365→  }
   366→}
   367→
   368→function formatDateTime(iso: string | null): string {
   369→  if (!iso) return '—'
   370→  try {
   371→    return new Date(iso).toLocaleString('fr-FR', {
   372→      day: '2-digit',
   373→      month: '2-digit',
   374→      year: 'numeric',
   375→      hour: '2-digit',
   376→      minute: '2-digit',
   377→    })
   378→  } catch {
   379→    return '—'
   380→  }
   381→}

   383→function toDateInput(iso: string | null): string {

   386→    return new Date(iso).toISOString().slice(0, 10)

   388→    return ''
   389→  }

   396→function DashboardTab({ fetchWithAuth }: { fetchWithAuth: (url: string, init?: RequestInit) => Promise<Response | null> }) {
   397→  const [data, setData] = useState<DashboardData | null>(null)
   398→  const [loading, setLoading] = useState(true)
   399→
   400→  const load = useCallback(async () => {
   401→    setLoading(true)
   402→    const res = await fetchWithAuth('/api/platform/dashboard')
   403→    if (!res) {
   404→      setLoading(false)
   405→      return
   406→    }
   407→    try {
   408→      const json = (await res.json()) as DashboardData
   409→      setData(json)
   410→    } catch {
   411→      toast.error('Réponse invalide du serveur')
   412→    } finally {
   413→      setLoading(false)
   414→    }
   415→  }, [fetchWithAuth])
   416→
   417→  useEffect(() => {
   418→    load()
   419→  }, [load])
   420→
   421→  if (loading) {
   422→    return (
   423→      <div className="space-y-6 p-4 md:p-6">
   424→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   425→          {Array.from({ length: 4 }).map((_, i) => (
   426→            <Skeleton key={i} className="h-28 rounded-xl" />

   430→          <Skeleton className="h-96 rounded-xl" />
   431→          <Skeleton className="h-96 rounded-xl" />

   433→      </div>

   437→  if (!data) return null
   438→

   764→function WeddingsTab({ fetchWithAuth }: { fetchWithAuth: (url: string, init?: RequestInit) => Promise<Response | null> }) {
   765→  const [weddings, setWeddings] = useState<Wedding[]>([])
   766→  const [total, setTotal] = useState(0)
   767→  const [totalPages, setTotalPages] = useState(1)
   768→  const [page, setPage] = useState(1)
   769→  const [loading, setLoading] = useState(true)
   770→  const [search, setSearch] = useState('')
   771→  const [statusFilter, setStatusFilter] = useState<string>('ALL')
   772→  const [planFilter, setPlanFilter] = useState<string>('ALL')
   773→
   774→  // Dialog state
   775→  const [showFormDialog, setShowFormDialog] = useState(false)
   776→  const [showDeleteDialog, setShowDeleteDialog] = useState(false)
   777→  const [editing, setEditing] = useState<Wedding | null>(null)
   778→  const [deleting, setDeleting] = useState<Wedding | null>(null)
   779→  const [form, setForm] = useState<WeddingFormState>(EMPTY_WEDDING_FORM)
   780→  const [saving, setSaving] = useState(false)
   781→
   782→  const LIMIT = 20
   783→  const searchRef = useRef(search)
   784→  searchRef.current = search
   785→
   786→  const load = useCallback(
   787→    async (targetPage: number) => {
   788→      setLoading(true)
   789→      const params = new URLSearchParams({
   790→        page: String(targetPage),
   791→        limit: String(LIMIT),
   792→      })
   793→      if (searchRef.current) params.set('search', searchRef.current)
   794→      if (statusFilter !== 'ALL') params.set('status', statusFilter)
   795→      if (planFilter !== 'ALL') params.set('plan', planFilter)
   796→
   797→      const res = await fetchWithAuth(`/api/platform/weddings?${params.toString()}`)
   798→      if (!res) {
   799→        setLoading(false)
   800→        return
   801→      }
   802→      try {
   803→        const json = (await res.json()) as PaginatedWeddings
   804→        setWeddings(json.weddings || [])
   805→        setTotal(json.total || 0)
   806→        // API returns { total, page, limit } — compute totalPages client-side.
   807→        setTotalPages(Math.max(1, Math.ceil((json.total || 0) / (json.limit || LIMIT))))
   808→        setPage(json.page || targetPage)
   809→      } catch {
   810→        toast.error('Réponse invalide du serveur')
   811→      } finally {
   812→        setLoading(false)
   813→      }

   815→    [fetchWithAuth, statusFilter, planFilter]

   819→    load(1)
   820→  }, [statusFilter, planFilter, load])
   821→

   823→  useEffect(() => {
   824→    const t = setTimeout(() => {
   825→      load(1)
   826→    }, 350)
   827→    return () => clearTimeout(t)
   828→  }, [search, load])

   831→    setEditing(null)
   832→    setForm(EMPTY_WEDDING_FORM)
   833→    setShowFormDialog(true)
   834→  }

   837→    setEditing(w)
   838→    setForm({
   839→      slug: w.slug,

   845→      plan: w.plan,
   846→    })

   848→  }
   849→
   850→  const handleSave = async () => {
   851→    if (!form.slug || !form.brideName || !form.groomName) {
   852→      toast.error('Slug, mariée et marié sont requis')
   853→      return
   854→    }
   855→    setSaving(true)
   856→    const payload: Record<string, unknown> = {
   857→      slug: form.slug.trim().toLowerCase(),
   858→      brideName: form.brideName.trim(),
   859→      groomName: form.groomName.trim(),
   860→      weddingDate: form.weddingDate ? new Date(form.weddingDate).toISOString() : null,
   861→      venueCity: form.venueCity.trim() || null,
   862→      status: form.status,
   863→      plan: form.plan,
   864→    }

   866→      const url = editing
   867→        ? `/api/platform/weddings/${editing.id}`
   868→        : '/api/platform/weddings'
   869→      const method = editing ? 'PUT' : 'POST'
   870→      const res = await fetchWithAuth(url, {
   871→        method,
   872→        headers: { 'Content-Type': 'application/json' },
   873→        body: JSON.stringify(payload),
   874→      })
   875→      if (!res) {
   876→        setSaving(false)
   877→        return
   878→      }
   879→      const json = await res.json().catch(() => ({}))
   880→      if (res.ok) {
   881→        toast.success(editing ? 'Mariage mis à jour' : 'Mariage créé')
   882→        setShowFormDialog(false)
   883→        setEditing(null)
   884→        setForm(EMPTY_WEDDING_FORM)
