   237→
   238→function formatUsd(cents: number): string {
   239→  return `$${(cents / 100).toFixed(2)}`
   240→}
   241→
   242→function formatFcfa(cents: number): string {
   243→  const usd = cents / 100
   244→  return `${Math.round(usd * FCFA_TO_USD_RATE).toLocaleString('fr-FR')} FCFA`
   245→}
   246→
   247→function formatDate(iso: string | null): string {
   248→  if (!iso) return '—'
   249→  return new Date(iso).toLocaleDateString('fr-FR', {
   250→    day: '2-digit',
   251→    month: 'short',
   252→    year: 'numeric',
   253→    hour: '2-digit',
   254→    minute: '2-digit',
   255→  })
   256→}

   259→// Main BillingTab component
   260→// ══════════════════════════════════════════════════════════════════════════════
   261→
   262→export function BillingTab({
   263→  fetchWithAuth,
   264→}: {
   265→  fetchWithAuth: (url: string, init?: RequestInit) => Promise<Response | null>
   266→}) {
   267→  const [data, setData] = useState<BillingOverviewResponse | null>(null)
   268→  const [loading, setLoading] = useState(true)
   269→  const [search, setSearch] = useState('')
   270→  const [statusFilter, setStatusFilter] = useState('ALL')
   271→  const [planFilter, setPlanFilter] = useState('ALL')
   272→
   273→  // Editor dialog state
   274→  const [editing, setEditing] = useState<BillingWeddingRow | null>(null)
   275→  const [editorOpen, setEditorOpen] = useState(false)
   276→  const [subscriptionDetail, setSubscriptionDetail] = useState<SubscriptionSummary | null>(null)
   277→  const [invoices, setInvoices] = useState<InvoiceRow[]>([])
   278→  const [form, setForm] = useState<SubscriptionFormState>(EMPTY_FORM)
   279→  const [saving, setSaving] = useState(false)
   280→  const [creatingInvoice, setCreatingInvoice] = useState(false)
   281→  const [actionLoading, setActionLoading] = useState<string | null>(null) // invoice id being acted upon
   282→
   283→  // WhatsApp modal state
   284→  const [whatsappData, setWhatsappData] = useState<WhatsAppResponse | null>(null)
   285→  const [whatsappOpen, setWhatsappOpen] = useState(false)
   286→  const [generatingWhatsapp, setGeneratingWhatsapp] = useState(false)
   287→
   288→  const searchRef = useRef(search)
   289→  searchRef.current = search
   290→
   291→  const load = useCallback(async () => {
   292→    setLoading(true)
   293→    const params = new URLSearchParams()
   294→    if (searchRef.current) params.set('search', searchRef.current)
   295→    if (statusFilter !== 'ALL') params.set('status', statusFilter)
   296→    if (planFilter !== 'ALL') params.set('plan', planFilter)
   297→    const res = await fetchWithAuth(`/api/platform/billing/weddings?${params.toString()}`)
   298→    if (!res) {
   299→      setLoading(false)
   300→      return
   301→    }
   302→    try {
   303→      const json = (await res.json()) as BillingOverviewResponse
   304→      setData(json)
   305→    } catch {
   306→      toast.error('Réponse invalide du serveur')
   307→    } finally {
   308→      setLoading(false)
   309→    }
   310→  }, [fetchWithAuth, statusFilter, planFilter])
   311→
   312→  useEffect(() => {
   313→    load()
   314→  }, [statusFilter, planFilter, load])
   315→
   316→  // Debounced search
   317→  useEffect(() => {
   318→    const t = setTimeout(() => load(), 350)
   319→    return () => clearTimeout(t)
   320→  }, [search, load])
   321→
   322→  // ─── Open subscription editor for a wedding ─────────────────────────────
   323→  const openEditor = async (row: BillingWeddingRow) => {
   324→    setEditing(row)
   325→    setEditorOpen(true)
   326→    setForm(EMPTY_FORM)
   327→    setSubscriptionDetail(null)
   328→    setInvoices([])
   329→
   330→    // Fetch current subscription detail + invoices in parallel
   331→    const [subRes, invRes] = await Promise.all([
   332→      fetchWithAuth(`/api/platform/weddings/${row.id}/subscription`),
   333→      fetchWithAuth(`/api/platform/weddings/${row.id}/invoices`),
   334→    ])
   335→    if (subRes) {
   336→      try {
   337→        const json = (await subRes.json()) as SubscriptionDetailResponse
   338→        if (json.subscription) {
   339→          setSubscriptionDetail(json.subscription)
   340→          setForm({
   341→            plan: json.subscription.plan,
   342→            status: json.subscription.status,
   343→            billingCycle: json.subscription.billingCycle,
   344→            amountAgreed:
   345→              json.subscription.amountAgreed != null
   346→                ? String(json.subscription.amountAgreed)
   347→                : '',
   348→            paymentMethod: json.subscription.paymentMethod || '',
   349→            whatsappPhone: json.subscription.whatsappPhone || '',
   350→            notes: json.subscription.notes || '',
   351→          })
   352→        } else {
   353→          // No subscription yet — default to wedding's current plan
   354→          setForm({
   355→            ...EMPTY_FORM,
   356→            plan: row.plan,
   357→          })
   358→        }
   359→      } catch {
   360→        /* ignore */
   361→      }
   362→    }
   363→    if (invRes) {
   364→      try {
   365→        const json = (await invRes.json()) as InvoicesListResponse
   366→        setInvoices(json.invoices || [])
   367→      } catch {
   368→        /* ignore */
   369→      }
   370→    }
   371→  }
   372→
   373→  // ─── Save subscription ──────────────────────────────────────────────────
   374→  const handleSaveSubscription = async () => {
   375→    if (!editing) return
   376→    setSaving(true)
   377→    const payload: Record<string, unknown> = {
   378→      plan: form.plan,
   379→      status: form.status,
   380→      billingCycle: form.billingCycle,
   381→      paymentMethod: form.paymentMethod || null,
   382→      whatsappPhone: form.whatsappPhone.trim() || null,
   383→      notes: form.notes.trim() || null,
   384→    }
   385→    if (form.amountAgreed.trim()) {
   386→      const n = Number(form.amountAgreed)
   387→      if (!Number.isFinite(n) || n < 0) {
   388→        toast.error('Prix personnalisé invalide (entier en centimes USD)')
   389→        setSaving(false)
   390→        return
   391→      }
   392→      payload.amountAgreed = Math.round(n)
   393→    } else {
   394→      payload.amountAgreed = null
   395→    }
   396→
   397→    const res = await fetchWithAuth(
   398→      `/api/platform/weddings/${editing.id}/subscription`,
   399→      {
   400→        method: 'PUT',
   401→        headers: { 'Content-Type': 'application/json' },
   402→        body: JSON.stringify(payload),
   403→      },
   404→    )
   405→    setSaving(false)
   406→    if (!res) return
   407→    if (res.ok) {
   408→      toast.success('Abonnement enregistré')
   409→      const json = (await res.json()) as { subscription: SubscriptionSummary }
   410→      setSubscriptionDetail(json.subscription)
   411→      await load() // refresh overview
   412→    } else {
   413→      const json = await res.json().catch(() => ({}))
   414→      toast.error(json.error || 'Erreur lors de la sauvegarde')
   415→    }
   416→  }
   417→
   418→  // ─── Generate WhatsApp message ──────────────────────────────────────────
   419→  const handleGenerateWhatsApp = async () => {
   420→    if (!editing) return
   421→    setGeneratingWhatsapp(true)
   422→    const res = await fetchWithAuth(
   423→      `/api/platform/weddings/${editing.id}/subscription/whatsapp`,
   424→      {
   425→        method: 'POST',
   426→        headers: { 'Content-Type': 'application/json' },
   427→        body: JSON.stringify({
   428→          plan: form.plan,
   429→          billingCycle: form.billingCycle,
   430→          whatsappPhone: form.whatsappPhone.trim() || null,
   431→          notes: form.notes.trim() || null,
   432→        }),
   433→      },
   434→    )
   435→    setGeneratingWhatsapp(false)
   436→    if (!res) return
   437→    if (res.ok) {
   438→      const json = (await res.json()) as WhatsAppResponse
   439→      setWhatsappData(json)
   440→      setWhatsappOpen(true)
   441→    } else {
   442→      const json = await res.json().catch(() => ({}))
   443→      toast.error(json.error || 'Erreur lors de la génération du message')
   444→    }
   445→  }
   446→
   447→  // ─── Create invoice ─────────────────────────────────────────────────────
   448→  const handleCreateInvoice = async () => {
   449→    if (!editing) return
   450→    setCreatingInvoice(true)
   451→    const res = await fetchWithAuth(
   452→      `/api/platform/weddings/${editing.id}/invoices`,
   453→      {
   454→        method: 'POST',
   455→        headers: { 'Content-Type': 'application/json' },
   456→        body: JSON.stringify({
   457→          plan: form.plan,
   458→          billingCycle: form.billingCycle,
   459→          paymentMethod: form.paymentMethod || null,
   460→          whatsappPhone: form.whatsappPhone.trim() || null,
   461→          notes: form.notes.trim() || null,
   462→        }),
   463→      },
   464→    )
   465→    setCreatingInvoice(false)
   466→    if (!res) return
   467→    if (res.ok) {
   468→      toast.success('Facture créée')
   469→      const json = (await res.json()) as { invoice: InvoiceRow }
   470→      setInvoices((prev) => [json.invoice, ...prev])
   471→      await load()
   472→    } else {
   473→      const json = await res.json().catch(() => ({}))
   474→      toast.error(json.error || 'Erreur lors de la création de la facture')
   475→    }
   476→  }
   477→
   478→  // ─── Mark invoice as paid ───────────────────────────────────────────────
   479→  const handleMarkPaid = async (invoice: InvoiceRow) => {
   480→    if (!editing) return
   481→    setActionLoading(invoice.id)
   482→    const res = await fetchWithAuth(`/api/platform/invoices/${invoice.id}`, {
   483→      method: 'PUT',
   484→      headers: { 'Content-Type': 'application/json' },
   485→      body: JSON.stringify({
   486→        status: 'PAID',
   487→        paymentMethod: invoice.paymentMethod || form.paymentMethod || null,
   488→      }),
   489→    })
   490→    setActionLoading(null)
   491→    if (!res) return
   492→    if (res.ok) {
   493→      toast.success('Facture marquée comme payée')
   494→      const json = (await res.json()) as { invoice: InvoiceRow }
   495→      setInvoices((prev) =>
   496→        prev.map((inv) => (inv.id === invoice.id ? json.invoice : inv)),
   497→      )
   498→      await load()
   499→    } else {
   500→      const json = await res.json().catch(() => ({}))
   501→      toast.error(json.error || 'Erreur lors du marquage')
   502→    }
   503→  }
   504→
   505→  // ─── Void invoice ───────────────────────────────────────────────────────
   506→  const handleVoidInvoice = async (invoice: InvoiceRow) => {
   507→    if (!editing) return
   508→    setActionLoading(invoice.id)
   509→    const res = await fetchWithAuth(`/api/platform/invoices/${invoice.id}`, {
   510→      method: 'PUT',
   511→      headers: { 'Content-Type': 'application/json' },
   512→      body: JSON.stringify({ status: 'VOID' }),
   513→    })
   514→    setActionLoading(null)
   515→    if (!res) return
   516→    if (res.ok) {
   517→      toast.success('Facture annulée')
   518→      const json = (await res.json()) as { invoice: InvoiceRow }
   519→      setInvoices((prev) =>
   520→        prev.map((inv) => (inv.id === invoice.id ? json.invoice : inv)),
   521→      )
   522→      await load()
   523→    } else {
   524→      const json = await res.json().catch(() => ({}))
   525→      toast.error(json.error || 'Erreur lors de l\'annulation')
   526→    }
   527→  }
   528→
   529→  // ─── Copy WhatsApp message to clipboard ─────────────────────────────────
   530→  const handleCopyMessage = async () => {
   531→    if (!whatsappData) return
   532→    try {
   533→      await navigator.clipboard.writeText(whatsappData.message)
   534→      toast.success('Message copié dans le presse-papiers')
   535→    } catch {
   536→      toast.error('Impossible de copier le message')
   537→    }
   538→  }
   539→
   540→  // ════════════════════════════════════════════════════════════════════════════
   541→  // Render
   542→  // ════════════════════════════════════════════════════════════════════════════
   543→
   544→  const summary = data?.summary
   545→
   546→  return (
   547→    <div className="space-y-6">
   548→      {/* ─── Summary cards ─────────────────────────────────────────────── */}
   549→      <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-5 gap-3">
   550→        <SummaryCard
   551→          icon={<Wallet className="w-4 h-4" />}
   552→          label="Total mariages"
   553→          value={summary?.total ?? 0}
   554→          loading={loading}
   555→        />
   556→        <SummaryCard
   557→          icon={<CheckCircle2 className="w-4 h-4" />}
   558→          label="Actifs"
   559→          value={summary?.active ?? 0}
   560→          loading={loading}
   561→          accent="emerald"
   562→        />
   563→        <SummaryCard
   564→          icon={<Clock className="w-4 h-4" />}
   565→          label="En attente"
   566→          value={summary?.pending ?? 0}
   567→          loading={loading}
   568→          accent="amber"
   569→        />
   570→        <SummaryCard
   571→          icon={<TrendingUp className="w-4 h-4" />}
   572→          label="MRR (USD)"
   573→          value={`$${(summary?.mrrUsd ?? 0).toFixed(2)}`}
   574→          loading={loading}
   575→          accent="gold"
   576→        />
   577→        <SummaryCard
   578→          icon={<DollarSign className="w-4 h-4" />}
   579→          label="À recouvrer"
   580→          value={`$${(summary?.pendingUsd ?? 0).toFixed(2)}`}
   581→          loading={loading}
   582→          accent="orange"
   583→        />
   584→      </div>
