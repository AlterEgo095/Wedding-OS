   286→// usePlatformFetch — wraps fetch with Bearer token + session-expiry handling
   287→// ════════════════════════════════════════════════════════════════════════════
   288→
   289→function usePlatformFetch() {
   290→  const router = useRouter()
   291→  const sessionExpiredRef = useRef(false)
   292→
   293→  const onSessionExpired = useCallback(() => {
   294→    if (sessionExpiredRef.current) return
   295→    sessionExpiredRef.current = true
   296→    try {
   297→      localStorage.removeItem('admin_token')
   298→      localStorage.removeItem('admin_user')
   299→    } catch {
   300→      /* ignore */
   301→    }
   302→    toast.error('Session expirée, veuillez vous reconnecter')
   303→    router.replace('/platform/login')
   304→  }, [router])
   305→
   306→  const getToken = useCallback(() => {
   307→    if (typeof window === 'undefined') return null
   308→    return localStorage.getItem('admin_token')
   309→  }, [])
   310→
   311→  const fetchWithAuth = useCallback(
   312→    async (url: string, init?: RequestInit): Promise<Response | null> => {
   313→      const token = getToken()
   314→      if (!token) {
   315→        onSessionExpired()
   316→        return null
   317→      }
   318→      let res: Response
   319→      try {
   320→        res = await fetch(url, {
   321→          ...init,
   322→          headers: {
   323→            ...(init?.headers || {}),
   324→            Authorization: `Bearer ${token}`,
   325→          },
   326→        })
   327→      } catch {
   328→        toast.error('Erreur de connexion au serveur')
   329→        return null
   330→      }
   331→      if (res.status === 401) {
   332→        onSessionExpired()
   333→        return null
   334→      }
   335→      if (res.status === 403) {
   336→        toast.error('Accès refusé')
   337→        return null
   338→      }
   339→      return res
   340→    },
   341→    [getToken, onSessionExpired]
   342→  )
   343→
   344→  return { fetchWithAuth, onSessionExpired, getToken }
   345→}

   375→function formatDate(iso: string | null): string {
   376→  if (!iso) return '—'
   377→  try {
   378→    return new Date(iso).toLocaleDateString('fr-FR', {
   379→      day: '2-digit',
   380→      month: 'short',
   381→      year: 'numeric',
   382→    })
   383→  } catch {
   384→    return '—'
   385→  }
   386→}

   388→function formatDateTime(iso: string | null): string {
   389→  if (!iso) return '—'
   390→  try {
   391→    return new Date(iso).toLocaleString('fr-FR', {
   392→      day: '2-digit',
   393→      month: '2-digit',
   394→      year: 'numeric',
   395→      hour: '2-digit',
   396→      minute: '2-digit',
   397→    })
   398→  } catch {
   399→    return '—'
   400→  }
   401→}

   416→function DashboardTab({ fetchWithAuth }: { fetchWithAuth: (url: string, init?: RequestInit) => Promise<Response | null> }) {
   417→  const [data, setData] = useState<DashboardData | null>(null)
   418→  const [loading, setLoading] = useState(true)
   419→
   420→  const load = useCallback(async () => {
   421→    setLoading(true)
   422→    const res = await fetchWithAuth('/api/platform/dashboard')
   423→    if (!res) {
   424→      setLoading(false)
   425→      return
   426→    }
   427→    try {
   428→      const json = (await res.json()) as DashboardData
   429→      setData(json)
   430→    } catch {
   431→      toast.error('Réponse invalide du serveur')
   432→    } finally {
   433→      setLoading(false)
   434→    }
   435→  }, [fetchWithAuth])
   436→
   437→  useEffect(() => {
   438→    load()
   439→  }, [load])

   755→      </div>
