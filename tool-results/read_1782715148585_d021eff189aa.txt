   343→
   344→function formatUsd(cents: number): string {
   345→  return `$${(cents / 100).toFixed(2)}`
   346→}
   347→
   348→function formatFcfa(cents: number): string {
   349→  return `${usdCentsToFcfa(cents).toLocaleString('fr-FR')} FCFA`
   350→}
   351→
   352→function formatDate(iso: string | null): string {
   353→  if (!iso) return '—'
   354→  return new Date(iso).toLocaleDateString('fr-FR', {
   355→    day: '2-digit',
   356→    month: 'short',
   357→    year: 'numeric',
   358→  })
   359→}
   360→
   361→function toDateInput(iso: string | null): string {
   362→  if (!iso) return ''
   363→  const d = new Date(iso)
   364→  if (Number.isNaN(d.getTime())) return ''
   365→  return d.toISOString().slice(0, 10)
   366→}
   367→
   368→/** Generate a random 12-character password (mixed alphanumeric). */
   369→function generateRandomPassword(length = 12): string {
   370→  const chars = 'abcdefghjkmnpqrstuvwxyzABCDEFGHJKLMNPQRSTUVWXYZ23456789'
   371→  let out = ''
   372→  const arr = new Uint32Array(length)
   373→  crypto.getRandomValues(arr)
   374→  for (let i = 0; i < length; i++) {
   375→    out += chars[arr[i] % chars.length]
   376→  }
   377→  return out
   378→}

   397→    NEW: 0,
   398→    CONTACTED: 0,
   399→    CONVERTED: 0,
   400→    REJECTED: 0,
   401→  })
   402→
   403→  // Wizard state
   404→  const [wizardOpen, setWizardOpen] = useState(false)
   405→  const [wizardStep, setWizardStep] = useState(1)
   406→  const [form, setForm] = useState<WizardFormState>(EMPTY_FORM)
   407→  const [leadId, setLeadId] = useState<string | null>(null)
   408→  const [sourceLead, setSourceLead] = useState<Lead | null>(null)
   409→  const [submitting, setSubmitting] = useState(false)
   410→  const [success, setSuccess] = useState<CreateWeddingResponse | null>(null)
   411→  const [slugChecking, setSlugChecking] = useState(false)
   412→  const [slugStatus, setSlugStatus] = useState<'idle' | 'available' | 'taken' | 'invalid'>('idle')
   413→
   414→  // Notes dialog
   415→  const [notesLead, setNotesLead] = useState<Lead | null>(null)
   416→  const [notesValue, setNotesValue] = useState('')
   417→  const [notesOpen, setNotesOpen] = useState(false)
   418→  const [notesSaving, setNotesSaving] = useState(false)
   419→
   420→  // Reject confirmation
   421→  const [rejectLead, setRejectLead] = useState<Lead | null>(null)
   422→
   423→  const LIMIT = 20
   424→  const searchRef = useRef(search)
   425→  searchRef.current = search
   426→
   427→  // ─── Fetch leads ────────────────────────────────────────────────────────
   428→  const load = useCallback(
   429→    async (targetPage: number) => {
   430→      setLoading(true)
   431→      const params = new URLSearchParams({
   432→        page: String(targetPage),
   433→        limit: String(LIMIT),
   434→      })
   435→      if (searchRef.current) params.set('search', searchRef.current)
   436→      if (statusFilter !== 'ALL') params.set('status', statusFilter)
   437→
   438→      const res = await fetchWithAuth(`/api/onboarding/leads?${params.toString()}`)
   439→      if (!res) {
   440→        setLoading(false)
   441→        return
   442→      }
   443→      try {
   444→        const json = (await res.json()) as LeadsListResponse
   445→        setLeads(json.leads || [])
   446→        setTotal(json.total || 0)
   447→        setTotalPages(Math.max(1, Math.ceil((json.total || 0) / (json.limit || LIMIT))))
   448→        setPage(json.page || targetPage)
   449→        setSummary(json.summary || { NEW: 0, CONTACTED: 0, CONVERTED: 0, REJECTED: 0 })
   450→      } catch {
   451→        toast.error('Réponse invalide du serveur')
   452→      } finally {
   453→        setLoading(false)
   454→      }
   455→    },
   456→    [fetchWithAuth, statusFilter],
   457→  )
   458→
   459→  useEffect(() => {
   460→    load(1)
   461→  }, [statusFilter, load])
   462→
   463→  // Debounced search (300ms)
   464→  useEffect(() => {
   465→    const t = setTimeout(() => load(1), 300)
   466→    return () => clearTimeout(t)
   467→  }, [search, load])

   471→    async (lead: Lead, patch: { status?: LeadStatus; notes?: string | null }) => {

   473→        method: 'PATCH',
   474→        headers: { 'Content-Type': 'application/json' },
   475→        body: JSON.stringify(patch),
   476→      })

   478→      if (res.ok) {
   479→        const json = (await res.json()) as { lead: Lead }
   480→        setLeads((prev) => prev.map((l) => (l.id === lead.id ? json.lead : l)))
   481→        // Refresh summary counts (cheapest path: reload)
   482→        load(page)
   483→        return true
   484→      }
   485→      const json = await res.json().catch(() => ({}))
   486→      toast.error(json.error || 'Erreur lors de la mise à jour du lead')

   488→    },
   489→    [fetchWithAuth, load, page],

   529→    setSourceLead(null)

   534→
   535→  const openWizardFromLead = useCallback((lead: Lead) => {

   538→      brideName: lead.brideName,

   543→      whatsappPhone: lead.phone || '',

   546→      slug: generateSlug(lead.brideName, lead.groomName),
   547→      slugTouched: false,
   548→    })
   549→    setLeadId(lead.id)
   550→    setSourceLead(lead)
   551→    setWizardStep(1)
   552→    setSlugStatus('idle')
   553→    setWizardOpen(true)
   554→  }, [])
   555→
   556→  // Auto-suggest slug when bride/groom names change (only if user hasn't manually edited)
   557→  useEffect(() => {
   558→    if (!wizardOpen) return
   559→    if (form.slugTouched) return
   560→    if (!form.brideName.trim() && !form.groomName.trim()) return
   561→    const suggested = generateSlug(form.brideName, form.groomName)
   562→    if (suggested !== form.slug) {
   563→      setForm((f) => ({ ...f, slug: suggested }))
   564→      setSlugStatus('idle')

   567→
   568→  // ─── Slug availability check ─────────────────────────────────────────────
   569→  const checkSlugAvailability = useCallback(async () => {
   570→    const slug = form.slug.trim().toLowerCase()
   571→    if (!slug) {
   572→      toast.error('Veuillez saisir un slug')
   573→      return
   574→    }

   576→      setSlugStatus('invalid')
   577→      return
   578→    }
   579→    setSlugChecking(true)
   580→    const res = await fetchWithAuth(`/api/platform/weddings?search=${encodeURIComponent(slug)}&limit=50`)
   581→    setSlugChecking(false)
   582→    if (!res) return
   583→    try {
   584→      const json = (await res.json()) as { weddings: Array<{ slug: string }> }
   585→      const exists = (json.weddings || []).some((w) => w.slug.toLowerCase() === slug)
   586→      setSlugStatus(exists ? 'taken' : 'available')
   587→      if (!exists) toast.success(`Slug "${slug}" disponible`)

   590→      toast.error('Impossible de vérifier le slug')
   591→    }
   592→  }, [fetchWithAuth, form.slug])
   593→
   594→  // ─── Step validation ─────────────────────────────────────────────────────

   601→        return 'Slug invalide (3-32 caractères, minuscules, chiffres et traits d\'union)'
   602→      }
